#ifndef BITS_H
#define BITS_H
#include <stdio.h>
#include <stdint.h>

static inline void print_bin8(const char *label, uint8_t v) {
    printf("%-12s 0b", label);
    for (int i = 7; i >= 0; i--) putchar((v >> i) & 1 ? '1' : '0');
    printf("  0x%02X  %3u\n", v, v);
}

static inline void print_bin32(const char *label, uint32_t v) {
    printf("%-12s 0b", label);
    for (int i = 31; i >= 0; i--) { putchar((v >> i) & 1 ? '1' : '0'); if (i % 8 == 0 && i) putchar('_'); }
    printf("  0x%08X\n", v);
}
#endif

🎬 D3-1 구조체 메모리 뷰어 — 필드가 메모리에 놓이는 순서와 복사

▶ 예제 1-1 · 어제의 Reading, 다시 보기

#include <stdio.h>

typedef struct {
    double t;      // 온도
    int    h;      // 습도
    int    d;      // 거리(mm)
} Reading;

int main(void) {
    Reading r = {25.3, 60, 120};          // 순서대로 초기화
    Reading s = {.d = 150, .t = 36.9};    // 지정 초기화 (C99) — 빠진 h 는 0

    printf("%.1f %d %d\n", r.t, r.h, r.d);    // → 25.3 60 120
    printf("%.1f %d %d\n", s.t, s.h, s.d);    // → 36.9 0 150

    Reading c = r;                        // 통째로 복사 (memcpy 와 같음)
    c.t = 99.0;
    printf("%.1f %.1f\n", r.t, c.t);      // → 25.3 99.0   (원본은 그대로)

    Reading *p = &r;
    printf("%d %d\n", p->h, (*p).h);      // → 60 60     (-> 는 (*p). 의 줄임)
    return 0;
}

어제 8교시 정렬에서 Reading t = a[j]; a[j] = a[j+1]; a[j+1] = t; 가 동작한 이유가 "구조체 대입은 통째 복사" 입니다. 배열은 = 로 복사되지 않지만 구조체는 됩니다 — 구조체 안의 배열까지.

▶ 예제 1-2 · 값으로 넘기기 vs 포인터로 넘기기

#include <stdio.h>

typedef struct { double t; int h; int d; } Reading;

void bump_by_value(Reading r)   { r.t += 1.0; }          // 복사본 수정 — 원본 불변
void bump_by_ptr(Reading *r)    { r->t += 1.0; }         // 원본 수정
double avg_t(const Reading *a, int n) {                  // 읽기만: const 포인터
    double s = 0;
    for (int i = 0; i < n; i++) s += a[i].t;             // a[i].t == (a + i)->t
    return s / n;
}

int main(void) {
    Reading r = {25.0, 60, 120};
    bump_by_value(r);  printf("%.1f\n", r.t);    // → 25.0
    bump_by_ptr(&r);   printf("%.1f\n", r.t);    // → 26.0

    Reading log[3] = {{24.0, 1, 1}, {26.0, 2, 2}, {28.0, 3, 3}};
    printf("%.1f\n", avg_t(log, 3));             // → 26.0
    return 0;
}

값으로 넘기면 16바이트가 복사됩니다. 구조체가 크면 느리고, 수정도 안 되므로 함수 인자는 포인터(const Reading *)가 기본입니다. Day 2 의 "원본을 바꾸려면 주소" 규칙 그대로.

▶ 예제 1-3 · ⚠ 함정 — 구조체 비교는 == 로 안 된다

#include <stdio.h>
#include <string.h>

typedef struct { double t; int h; int d; } Reading;

int same(const Reading *a, const Reading *b) {
    return a->t == b->t && a->h == b->h && a->d == b->d;   // 필드별 비교
}

int main(void) {
    Reading a = {25.3, 60, 120}, b = {25.3, 60, 120};
    // if (a == b) ...          // error: invalid operands to binary ==
    printf("%d\n", same(&a, &b));                        // → 1
    printf("%d\n", memcmp(&a, &b, sizeof a) == 0);       // → 보통 1 — 하지만 패딩 때문에 믿으면 안 됨 (2교시)
    return 0;
}

🎬 D3-2 패딩 실험실 — 필드를 끌어다 순서를 바꾸면 크기가 변한다

▶ 예제 2-1 · 12 가 아니라 16

#include <stdio.h>
#include <stddef.h>          // offsetof

typedef struct { char id; int v; char f; } Bad;     // 1 + 4 + 1 = 6 ?
typedef struct { int v; char id; char f; } Good;    // 4 + 1 + 1 = 6 ?
typedef struct { double t; int h; int d; } Reading; // 8 + 4 + 4 = 16

int main(void) {
    printf("Bad  %zu  (id@%zu v@%zu f@%zu)\n", sizeof(Bad),  offsetof(Bad, id),  offsetof(Bad, v),  offsetof(Bad, f));
    printf("Good %zu  (v@%zu id@%zu f@%zu)\n", sizeof(Good), offsetof(Good, v), offsetof(Good, id), offsetof(Good, f));
    printf("Reading %zu\n", sizeof(Reading));
    return 0;
}

Bad  12  (id@0 v@4 f@8)
Good 8   (v@0 id@4 f@5)
Reading 16

규칙 두 가지: ① 각 필드는 자기 크기의 배수 주소에서 시작한다 (int 는 4 의 배수, double 은 8 의 배수). ② 구조체 전체 크기는 가장 큰 필드 크기의 배수다 (배열로 늘어놓아도 ①이 유지되도록). Bad 는 id 뒤에 3바이트, f 뒤에 3바이트 패딩. 같은 필드를 큰 것부터 배치하면 패딩이 사라집니다.

▶ 예제 2-2 · 패딩을 없애는 두 방법과 대가

#include <stdio.h>
#include <stdint.h>

#pragma pack(push, 1)                            // 이 아래는 패딩 금지
typedef struct { char id; int v; char f; } Packed;
#pragma pack(pop)

typedef struct { uint16_t dist; int8_t temp; uint8_t hum; } Frame;   // Day 1 4교시의 그 구조체

int main(void) {
    printf("%zu %zu\n", sizeof(Packed), sizeof(Frame));   // → 6 4
    return 0;
}
방법 장점 대가
필드 순서를 큰 것부터 이식성 100%, 성능 손실 없음 설계 시 신경 써야 함
#pragma pack(1) 통신 프레임·파일 포맷과 1:1 대응 정렬 안 된 접근 → x86 은 느려지고, 일부 MCU 는 죽음 (Day 2 2교시 경고)

프레임을 구조체로 바로 캐스팅하고 싶을 때((Frame *)buf) 이 문제가 나옵니다. 안전한 답은 Day 2 방법 ②: 바이트 조립. 4교시에서 완성.

▶ 예제 2-3 · union — 같은 4바이트를 세 가지로 읽기

#include <stdio.h>
#include <stdint.h>

typedef union {
    uint32_t w;
    uint8_t  b[4];
    float    f;
} Word;

int main(void) {
    Word x;
    x.w = 0x11223344;
    printf("%02X %02X %02X %02X\n", x.b[0], x.b[1], x.b[2], x.b[3]);   // → 44 33 22 11 (리틀 엔디안)
    x.f = 1.0f;
    printf("0x%08X\n", x.w);                                           // → 0x3F800000 (IEEE 754)
    printf("%zu\n", sizeof(Word));                                     // → 4 (가장 큰 멤버)
    return 0;
}

union 은 멤버가 같은 자리를 공유합니다. 크기는 가장 큰 멤버. 센서가 float 을 4바이트로 보내올 때 b[] 에 받아 f 로 읽는 것이 전형적 용도. 마지막에 쓴 멤버만 의미가 있습니다.

▶ 예제 2-4 · enum — 상태에 이름을, 그리고 개수를

#include <stdio.h>

typedef enum { ST_IDLE, ST_HEAT, ST_COOL, ST_ALERT, ST_COUNT } State;   // 0,1,2,3,4
static const char *NAME[ST_COUNT] = {"IDLE", "HEAT", "COOL", "ALERT"};

int main(void) {
    State st = ST_COOL;
    printf("%d %s %zu\n", st, NAME[st], sizeof st);     // → 2 COOL 4
    for (int i = 0; i < ST_COUNT; i++) printf("%s ", NAME[i]);   // → IDLE HEAT COOL ALERT
    printf("\n");
    return 0;
}

마지막에 ST_COUNT 를 두는 관용구 — 상태를 추가해도 배열 크기와 루프 상한이 자동으로 맞습니다. 7교시 상태 머신 테이블의 인덱스가 이 enum 입니다. enum 은 그냥 int 라서 st = 99 도 컴파일됩니다 — switch 의 default 로 방어.

✏️ 빈칸 채우기 2-1

패딩 없이 8바이트가 되도록 필드 순서를 정하고, 비트 플래그 union 을 완성하세요.

#include <stdio.h>
#include <stdint.h>

typedef struct {                 // 순서를 바꿔 sizeof == 8 이 되게
     a;   // uint8_t
     b;   // uint32_t
     c;   // uint16_t
     d;   // uint8_t
} Rec;

typedef union {
    uint8_t raw;
    struct { uint8_t on:1, err:1, mode:2, level:; } bits;   // 총 8비트
} Flags;

int main(void) {
    printf("%zu\n", sizeof(Rec));                        // → 8
    Flags f = {.raw = 0b10010101};
    printf("%u %u %u %u\n", f.bits.on, f.bits.err, f.bits.mode, f.bits.level);   // → 1 0 1 9
    return 0;
}
🔒 정답 보기 2-1 
typedef struct { uint32_t b; uint16_t c; uint8_t a; uint8_t d; } Rec;   // 4+2+1+1 = 8
    struct { uint8_t on:1, err:1, mode:2, level:4; } bits;             // 1+1+2+4 = 8비트
비트 필드는 LSB 부터 채워집니다(gcc): `10010101` → on=1, err=0, mode=01=1, level=1001=9. 비트 필드의 배치는 컴파일러마다 달라 통신 프레임에는 쓰지 않고, 4교시의 마스크·시프트를 씁니다.

✏️ 괄호 넣기 2-2

  1. { char; int; char } 구조체가 12 바이트인 이유는 int 가 의 배수 주소에 와야 하기 때문이다.
  2. 구조체 전체 크기는 필드 크기의 배수로 맞춰진다.
  3. #pragma pack(1) 은 패딩을 없애지만 접근이 생겨 일부 MCU 에서 죽을 수 있다.
  4. union 의 크기는 멤버의 크기이고, 멤버들은 를 공유한다.
  5. enum 마지막에 ST_COUNT 를 두는 이유는 를 자동으로 얻기 위해서다.

🎬 D3-3 비트 연산기 — 두 피연산자를 비트 단위로 놓고 결과를 본다

▶ 예제 3-1 · 여섯 연산자

#include "bits.h"

int main(void) {
    uint8_t a = 0b10110010;                 // 178  (0b 접두는 gcc 확장, C23 표준)
    print_bin8("a",       a);
    print_bin8("a & 0x0F", a & 0x0F);       // 하위 4비트만 남김        → 0b00000010
    print_bin8("a | 0x01", a | 0x01);       // 최하위 비트 켬           → 0b10110011
    print_bin8("a ^ 0xFF", a ^ 0xFF);       // 전부 반전 (XOR 1 = NOT)  → 0b01001101
    print_bin8("~a",       (uint8_t)~a);    // 반전 — 캐스팅 필요 (아래 함정)
    print_bin8("a >> 4",   a >> 4);         // 상위 니블을 아래로       → 0b00001011 = 11
    printf("a << 1 = %d\n", a << 1);        // → 356 !!  (8비트가 아니라 int 로 계산됨)
    return 0;
}

a << 1 이 100 이 아니라 356 인 이유: Day 1 5교시 정수 승격. uint8_t 는 연산 전에 int 로 바뀌므로 잘리지 않습니다. 8비트로 자르려면 (uint8_t)(a << 1). ~a 도 같은 이유로 0xFFFFFF4D 가 되어 캐스팅이 필요합니다.

▶ 예제 3-2 · 네 가지 관용구

#include "bits.h"

int main(void) {
    uint32_t reg = 0;
    reg |=  (1u << 3);                print_bin32("set 3",   reg);   // 켜기:  OR
    reg |=  (1u << 5);                print_bin32("set 5",   reg);   //        0x28
    reg &= ~(1u << 3);                print_bin32("clear 3", reg);   // 끄기:  AND NOT   0x20
    reg ^=  (1u << 5);                print_bin32("toggle 5",reg);   // 토글:  XOR      0x00
    reg |=  (1u << 5);
    printf("bit5 = %d, bit3 = %d\n", (reg >> 5) & 1, (reg >> 3) & 1);   // 검사: shift + AND  → 1, 0
    return 0;
}
하고 싶은 것 코드 왜
n번 비트 켜기 reg \|= (1u << n) OR 은 1 을 이기지 못함
n번 비트 끄기 reg &= ~(1u << n) 그 자리만 0 인 마스크와 AND
n번 비트 토글 reg ^= (1u << n) XOR 1 은 반전
n번 비트 검사 (reg >> n) & 1 그 비트를 0번으로 내려서 본다

1u 의 u: 1 << 31 은 signed 오버플로(정의되지 않음). 레지스터는 항상 unsigned.

센서 프레임 D=120 의 120(0x78 = 0111 1000)을 상태 플래그 8비트로 정의합니다.

#include "bits.h"

#define FLAG_POWER   (1u << 7)     // 1 = 배터리
#define FLAG_ONLINE  (1u << 6)
#define FLAG_ALARM   (1u << 5)
#define FLAG_LOWBAT  (1u << 4)
#define LEVEL_MASK   0x0Fu         // 하위 4비트 = 신호 세기 0~15

int main(void) {

    print_bin8("D", d);
    printf("power=%d online=%d alarm=%d lowbat=%d level=%u\n",
           !!(d & FLAG_POWER), !!(d & FLAG_ONLINE), !!(d & FLAG_ALARM), !!(d & FLAG_LOWBAT), d & LEVEL_MASK);
    // → power=0 online=1 alarm=1 lowbat=1 level=8
    return 0;
}

▶ 예제 4-3 · XOR 체크섬 — 프레임이 깨졌는지

#include <stdio.h>
#include <stdint.h>
#include <string.h>

uint8_t checksum(const char *s) {              // '$' 와 '*' 사이의 XOR
    uint8_t cs = 0;
    for (; *s && *s != '*'; s++) cs ^= (uint8_t)*s;
    return cs;
}

int main(void) {
    const char *body = "T=25.3,H=60,D=120";
    printf("%02X\n", checksum(body));          // → 2C
    // 프레임에 붙일 때: "$T=25.3,H=60,D=120*2C"
    const char *rx = "T=25.3,H=61,D=120";      // 한 글자 깨짐
    printf("%02X\n", checksum(rx));            // → 4B ≠ 2C → 폐기
    return 0;
}

NMEA(GPS) 프레임이 정확히 이 방식($...*hh)입니다. XOR 은 같은 값을 두 번 하면 원래대로 돌아오는 성질(a ^ b ^ b == a)이 있어 체크섬·토글·간단한 암호화에 쓰입니다.

    op = add;                      // 함수 이름 = 함수 주소 (배열 이름이 감쇠하듯)
    printf("%d\n", op(2, 3));      // → 5    ((*op)(2,3) 과 같음)
    op = mul;
    printf("%d\n", op(2, 3));      // → 6
    printf("%p %p\n", (void *)add, (void *)op);   // → 같은 주소
    return 0;
}

읽는 법: int (*op)(int, int) — 괄호가 *op 를 먼저 묶습니다. 괄호를 빼면 int *op(int, int) = "int 포인터를 돌려주는 함수" 가 되어 전혀 다른 뜻. 복잡해지면 typedef int (*BinOp)(int, int); 로 이름을 붙입니다.

#include <stdio.h>
#include <stdlib.h>

typedef struct { double t; int h; int d; } Reading;

int cmp_temp(const void *a, const void *b) {            // qsort 가 요구하는 모양
    const Reading *x = a, *y = b;                        // void * → 원래 타입으로
    return (x->t > y->t) - (x->t < y->t);                // -1 / 0 / 1  (x->t - y->t 는 double 이라 안 됨)
}
int cmp_hum_desc(const void *a, const void *b) {
    return ((const Reading *)b)->h - ((const Reading *)a)->h;   // 내림차순: b - a
}

int main(void) {
    Reading log[] = {{26.0, 61, 118}, {24.8, 59, 121}, {37.2, 70, 150}};
    int n = sizeof log / sizeof log[0];

    qsort(log, n, sizeof log[0], cmp_temp);
    for (int i = 0; i < n; i++) printf("%.1f ", log[i].t);        // → 24.8 26.0 37.2
    printf("\n");
    qsort(log, n, sizeof log[0], cmp_hum_desc);
    for (int i = 0; i < n; i++) printf("%d ", log[i].h);           // → 70 61 59
    printf("\n");
    return 0;
}

#include <stdio.h>
#include <stdint.h>

typedef enum { ST_IDLE, ST_HEAT, ST_COOL, ST_ALERT, ST_COUNT } State;
static const char *NAME[ST_COUNT] = {"IDLE", "HEAT", "COOL", "ALERT"};

volatile uint32_t GPIO = 0;                       // 5교시: 실제로는 레지스터 주소
#define BIT_HEATER (1u << 0)
#define BIT_FAN    (1u << 1)
#define BIT_ALARM  (1u << 2)

static State on_idle(double t)  { GPIO &= ~(BIT_HEATER | BIT_FAN | BIT_ALARM);
                                  if (t < 18) return ST_HEAT; if (t > 28) return ST_COOL; return ST_IDLE; }
static State on_heat(double t)  { GPIO |= BIT_HEATER;   if (t >= 20) return ST_IDLE; return ST_HEAT; }
static State on_cool(double t)  { GPIO |= BIT_FAN;      if (t > 35) return ST_ALERT; if (t <= 26) return ST_IDLE; return ST_COOL; }
static State on_alert(double t) { GPIO |= BIT_ALARM | BIT_FAN; if (t <= 30) return ST_COOL; return ST_ALERT; }

typedef State (*Handler)(double);
static const Handler TABLE[ST_COUNT] = { on_idle, on_heat, on_cool, on_alert };   // enum 순서와 일치!

int main(void) {
    double temps[] = {25, 17, 19, 21, 29, 33, 36, 31, 29, 25};
    State st = ST_IDLE;
    for (size_t i = 0; i < sizeof temps / sizeof temps[0]; i++) {
        State next = TABLE[st](temps[i]);                  // 현재 상태의 핸들러 호출
        printf("T=%4.1f %-5s -> %-5s GPIO=0x%X\n", temps[i], NAME[st], NAME[next], GPIO);
        st = next;
    }
    return 0;
}
