1-2. libgpiod v2 객체 흐름

v2 는 "무엇을 · 어떻게 · 누가" 를 각각 객체로 만든 뒤 한 번에 요청한다.

gpiod_chip_open("/dev/gpiochip0")                 ① 칩 열기
        │
gpiod_line_settings  ─ 방향 · 초기값 · bias · edge · active-low   ② 어떻게
        │  add_line_settings(offsets = {17})
gpiod_line_config    ─ "17번 라인에 이 설정"                     ③ 어떤 라인에
        │
gpiod_request_config ─ consumer = "blink"                       ④ 누가
        │
gpiod_chip_request_lines()  ──▶  gpiod_line_request              ⑤ 커널에 요청 → 소유
        │
set_value / get_value / read_edge_events … → release()          ⑥ 사용 후 반납
객체 MCU(Arduino) 에서 비슷한 것 메모
gpiod_chip — /dev/gpiochip0 을 연 fd
gpiod_line_settings pinMode(pin, OUTPUT) 의 인자 여러 라인에 재사용 가능
gpiod_line_config — offset ↔ settings 묶음
gpiod_request_config — consumer 이름, 이벤트 버퍼 크기
gpiod_line_request 핀을 "가진" 상태 이것만 있으면 칩은 닫아도 된다

settings · line_config · request_config 는 요청을 만들기 위한 재료라서 요청이 끝나면 바로 free 해도 된다.

1-3. 예제 01 — LED 깜빡이기

배선은 Day 1 그대로다 (GPIO17 → 220Ω → LED → GND).

src/01_blink.c — 교육용으로 객체를 하나씩 직접 만든다.

/* 01_blink.c — libgpiod v2 로 LED 깜빡이기 (객체 흐름을 한 단계씩 직접 작성)
 * 배선: GPIO17 → 220Ω → LED → GND
 * 빌드: make 01_blink      실행: ./01_blink      종료: Ctrl+C
 */
#include <gpiod.h>
#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>

#define CHIP_PATH "/dev/gpiochip0"
#define LED_LINE  17

static volatile sig_atomic_t g_stop = 0;
static void on_sigint(int sig) { (void)sig; g_stop = 1; }

int main(void)
{
    unsigned int offset = LED_LINE;
    int ret = EXIT_FAILURE;

    /* Ctrl+C → g_stop = 1 → 루프 종료 → 아래에서 라인 반납 */
    struct sigaction sa;
    memset(&sa, 0, sizeof sa);
    sa.sa_handler = on_sigint;
    sigaction(SIGINT, &sa, NULL);

    /* ① 칩 열기 */
    struct gpiod_chip *chip = gpiod_chip_open(CHIP_PATH);
    if (!chip) { perror("gpiod_chip_open"); return EXIT_FAILURE; }

    /* ② 라인 하나에 줄 설정: 출력, 처음 값 0 */
    struct gpiod_line_settings *settings = gpiod_line_settings_new();
    gpiod_line_settings_set_direction(settings, GPIOD_LINE_DIRECTION_OUTPUT);
    gpiod_line_settings_set_output_value(settings, GPIOD_LINE_VALUE_INACTIVE);

    /* ③ 어떤 라인(offset)에 그 설정을 쓸지 묶기 */
    struct gpiod_line_config *line_cfg = gpiod_line_config_new();
    gpiod_line_config_add_line_settings(line_cfg, &offset, 1, settings);

    /* ④ 요청 전체에 대한 설정: consumer 이름 (gpioinfo 에 보임) */
    struct gpiod_request_config *req_cfg = gpiod_request_config_new();
    gpiod_request_config_set_consumer(req_cfg, "blink");

    /* ⑤ 커널에 요청 → 성공하면 이 프로세스가 라인의 소유자 */
    struct gpiod_line_request *req = gpiod_chip_request_lines(chip, req_cfg, line_cfg);
    if (!req) { perror("gpiod_chip_request_lines"); goto cleanup; }

    printf("GPIO%d 깜빡임 시작 (Ctrl+C 로 종료)\n", LED_LINE);
    int on = 0;
    while (!g_stop) {
        on = !on;
        gpiod_line_request_set_value(req, offset,
            on ? GPIOD_LINE_VALUE_ACTIVE : GPIOD_LINE_VALUE_INACTIVE);
        struct timespec ts = { 0, 500 * 1000000L };      /* 0.5초 */
        nanosleep(&ts, NULL);
    }

    gpiod_line_request_set_value(req, offset, GPIOD_LINE_VALUE_INACTIVE);   /* 끄고 나가기 */
    gpiod_line_request_release(req);                                       /* ⑥ 반납 */
    printf("\n라인 반납 완료\n");
    ret = EXIT_SUCCESS;

cleanup:
    gpiod_request_config_free(req_cfg);
    gpiod_line_config_free(line_cfg);
    gpiod_line_settings_free(settings);
    gpiod_chip_close(chip);
    return ret;
}

1-4. Ctrl+C 와 반납 — 공통 헤더 gpio_common.h

Ctrl+C 를 누르면 프로세스가 바로 죽는 게 기본이다. 그러면 LED 가 켜진 채로 남거나(상태 보장 안 됨) 정리 코드가 돌지 않는다. 그래서 SIGINT 를 잡아 깃발만 세우고, 루프가 끝난 뒤 LED 를 끄고 반납한다.

설정 효과
sa_handler = on_sigint Ctrl+C → g_stop = 1 만 하고 돌아옴
SA_RESTART 를 주지 않음 poll() · nanosleep() 이 EINTR 로 즉시 깨어나 루프가 끝난다
volatile sig_atomic_t 핸들러와 main 이 안전하게 공유하는 변수

02 부터는 이 반복 코드를 src/gpio_common.h 에 모아 두고 #include "gpio_common.h" 로 쓴다. Day 3~5 에서도 같은 헤더를 쓴다.

/* gpio_common.h — Day 2~5 공통 도우미 (libgpiod v2)
 *
 *   gpio_install_sigint()  Ctrl+C 를 받으면 g_stop = 1 (poll·sleep 을 깨움)
 *   gpio_request_*()       라인 요청을 한 줄로 — 실패하면 NULL
 *   now_ns()               CLOCK_MONOTONIC 나노초 (edge event 타임스탬프와 같은 시계)
 *   sleep_ms()             Ctrl+C 로 깨어날 수 있는 대기
 *
 * 규칙: 요청한 라인은 프로그램이 끝나기 전에 gpiod_line_request_release() 로 반납한다.
 */
#ifndef GPIO_COMMON_H
#define GPIO_COMMON_H

#include <gpiod.h>
#include <signal.h>
#include <stdio.h>
#include <string.h>
#include <time.h>
#include <errno.h>

#define GPIO_CHIP "/dev/gpiochip0"

static volatile sig_atomic_t g_stop = 0;

static void on_sigint(int sig) { (void)sig; g_stop = 1; }

/* SA_RESTART 를 주지 않는다 → poll()/nanosleep() 이 EINTR 로 돌아와 루프가 끝난다 */
static inline void gpio_install_sigint(void)
{
    struct sigaction sa;
    memset(&sa, 0, sizeof sa);
    sa.sa_handler = on_sigint;
    sigemptyset(&sa.sa_mask);
    sigaction(SIGINT, &sa, NULL);
    sigaction(SIGTERM, &sa, NULL);
}

static inline long long now_ns(void)
{
    struct timespec ts;
    clock_gettime(CLOCK_MONOTONIC, &ts);
    return (long long)ts.tv_sec * 1000000000LL + ts.tv_nsec;
}

static inline void sleep_ms(long ms)
{
    struct timespec ts = { ms / 1000, (ms % 1000) * 1000000L };
    nanosleep(&ts, NULL);            /* SIGINT 가 오면 남은 시간을 버리고 돌아온다 */
}

/* 공통 요청 함수: settings 하나를 offsets[0..n-1] 전부에 적용 */
static inline struct gpiod_line_request *
gpio_request(const char *consumer, const unsigned int *offsets, size_t n,
             struct gpiod_line_settings *settings)
{
    struct gpiod_chip *chip = NULL;
    struct gpiod_line_config *lcfg = NULL;
    struct gpiod_request_config *rcfg = NULL;
    struct gpiod_line_request *req = NULL;

    chip = gpiod_chip_open(GPIO_CHIP);
    if (!chip) { perror("gpiod_chip_open"); goto out; }

    lcfg = gpiod_line_config_new();
    if (!lcfg || gpiod_line_config_add_line_settings(lcfg, offsets, n, settings) < 0) {
        perror("line_config"); goto out;
    }
    rcfg = gpiod_request_config_new();
    if (!rcfg) { perror("request_config"); goto out; }
    gpiod_request_config_set_consumer(rcfg, consumer);

    req = gpiod_chip_request_lines(chip, rcfg, lcfg);
    if (!req)
        fprintf(stderr, "%s: 라인 요청 실패: %s\n", consumer, strerror(errno));
out:
    gpiod_request_config_free(rcfg);
    gpiod_line_config_free(lcfg);
    gpiod_chip_close(chip);          /* 요청은 칩을 닫아도 유지된다 */
    return req;
}

/* 출력 라인 요청 (초기값 inactive = 0) */
static inline struct gpiod_line_request *
gpio_request_output(const char *consumer, const unsigned int *offsets, size_t n)
{
    struct gpiod_line_settings *s = gpiod_line_settings_new();
    if (!s) return NULL;
    gpiod_line_settings_set_direction(s, GPIOD_LINE_DIRECTION_OUTPUT);
    gpiod_line_settings_set_output_value(s, GPIOD_LINE_VALUE_INACTIVE);
    struct gpiod_line_request *r = gpio_request(consumer, offsets, n, s);
    gpiod_line_settings_free(s);
    return r;
}

/* 입력 라인 요청: pull-up, edge 는 GPIOD_LINE_EDGE_NONE 이면 이벤트 없음 */
static inline struct gpiod_line_request *
gpio_request_input(const char *consumer, const unsigned int *offsets, size_t n,
                   enum gpiod_line_edge edge, unsigned long debounce_us)
{
    struct gpiod_line_settings *s = gpiod_line_settings_new();
    if (!s) return NULL;
    gpiod_line_settings_set_direction(s, GPIOD_LINE_DIRECTION_INPUT);
    gpiod_line_settings_set_bias(s, GPIOD_LINE_BIAS_PULL_UP);
    gpiod_line_settings_set_edge_detection(s, edge);
    if (debounce_us)
        gpiod_line_settings_set_debounce_period_us(s, debounce_us);
    struct gpiod_line_request *r = gpio_request(consumer, offsets, n, s);
    gpiod_line_settings_free(s);
    return r;
}

✏️ 괄호 넣기 1-2

  1. libgpiod v2 에서 커널에 라인을 요청하는 함수는 gpiod_chip_ 이다.
  2. 요청에 붙이는 이름(consumer)은 명령 출력에서 볼 수 있다.
  3. SIGINT 핸들러를 등록할 때 — 그래야 poll 이 깨어난다.
  4. 핸들러와 main 이 함께 쓰는 깃발 변수의 자료형은 volatile 이다.
🔒 정답 보기 1-2
  1. request_lines
  2. gpioinfo
  3. SA_RESTART 를 주지 않는다
  4. sig_atomic_t

gpiod_chip_request_lines() 가 v2 의 유일한 요청 함수입니다(v1 의 gpiod_line_request_output 은 없음). SA_RESTART 가 있으면 블로킹 호출이 시그널 후 자동 재시작되어 루프가 g_stop 을 확인하지 못합니다. sig_atomic_t 는 시그널 핸들러에서 안전하게 읽고 쓸 수 있는 정수형입니다.

2-2. 한 요청에 라인 여러 개

라인마다 요청을 따로 만들 수도 있지만, 같은 설정이면 offsets 배열로 한 번에 요청하는 것이 좋다. 그러면 set_values() 한 번으로 세 LED 를 동시에 바꾼다 (ioctl 1회).

함수 하는 일
gpiod_line_request_set_value(req, offset, v) 라인 하나
gpiod_line_request_set_values(req, v[]) 요청한 모든 라인 — v[] 순서 = 요청할 때 넘긴 offsets 순서
gpiod_line_request_set_values_subset(req, n, offsets, v) 일부만

2-3. 예제 02 — 신호등 상태표

상태를 표(배열) 로 적고 루프가 표를 따라가게 만든다. 순서를 바꾸고 싶으면 코드가 아니라 표만 고치면 된다.

src/02_traffic.c

/* 02_traffic.c — 라인 3개를 한 요청으로 묶어 신호등 상태표 재생
 * 배선: GPIO17 빨강 · GPIO27 노랑 · GPIO22 초록 (각각 220Ω → LED → GND)
 * 실행: ./02_traffic          ./02_traffic -l (active-low 로 요청해 보기)
 */
#include "gpio_common.h"
#include <stdlib.h>

static const unsigned int LINES[3] = { 17, 27, 22 };      /* R, Y, G */

/* 상태표: 각 행 = { R, Y, G, 유지 시간(ms) } */
static const struct { int r, y, g; int ms; const char *name; } STEPS[] = {
    { 0, 0, 1, 3000, "초록" },
    { 0, 1, 0, 1000, "노랑" },
    { 1, 0, 0, 3000, "빨강" },
    { 1, 1, 0, 1000, "빨강+노랑" },
};
#define NSTEPS (sizeof STEPS / sizeof STEPS[0])

int main(int argc, char **argv)
{
    int active_low = (argc > 1 && strcmp(argv[1], "-l") == 0);
    gpio_install_sigint();

    struct gpiod_line_settings *s = gpiod_line_settings_new();
    gpiod_line_settings_set_direction(s, GPIOD_LINE_DIRECTION_OUTPUT);
    gpiod_line_settings_set_output_value(s, GPIOD_LINE_VALUE_INACTIVE);
    gpiod_line_settings_set_active_low(s, active_low);   /* 논리값 1 = 전압 0V 로 뒤집기 */

    struct gpiod_line_request *req = gpio_request("traffic", LINES, 3, s);
    gpiod_line_settings_free(s);
    if (!req) return EXIT_FAILURE;

    printf("신호등 시작%s (Ctrl+C 로 종료)\n", active_low ? " — active-low" : "");
    for (size_t i = 0; !g_stop; i = (i + 1) % NSTEPS) {
        /* values[] 의 순서 = 요청할 때 넘긴 offsets 순서 (17, 27, 22) */
        enum gpiod_line_value v[3] = {
            STEPS[i].r ? GPIOD_LINE_VALUE_ACTIVE : GPIOD_LINE_VALUE_INACTIVE,
            STEPS[i].y ? GPIOD_LINE_VALUE_ACTIVE : GPIOD_LINE_VALUE_INACTIVE,
            STEPS[i].g ? GPIOD_LINE_VALUE_ACTIVE : GPIOD_LINE_VALUE_INACTIVE,
        };
        gpiod_line_request_set_values(req, v);           /* 세 줄을 한 번에 */
        printf("  %-10s R=%d Y=%d G=%d\n", STEPS[i].name, STEPS[i].r, STEPS[i].y, STEPS[i].g);
        sleep_ms(STEPS[i].ms);
    }

    enum gpiod_line_value off[3] = { 0, 0, 0 };
    gpiod_line_request_set_values(req, off);
    gpiod_line_request_release(req);
    printf("\n종료\n");
    return EXIT_SUCCESS;
}

2-4. active-low — 논리값과 전압을 뒤집기

./02_traffic -l 로 실행해 보자. 표는 그대로인데 켜져야 할 LED 가 꺼지고, 꺼져야 할 LED 가 켜진다.

설정 ACTIVE (1) 을 쓰면 INACTIVE (0) 을 쓰면 쓰는 곳
active-high (기본) 3.3V 0V LED 를 GPIO → GND 로 연결 (오늘 배선)
active-low 0V 3.3V LED 를 3V3 → GPIO 로 연결, Low 에서 동작하는 모듈

active-low 를 쓰면 코드에서는 항상 "1 = 켜짐" 으로 생각하고, 실제 전압 반전은 커널이 해 준다. gpioinfo 에서 active-low 표시로 확인할 수 있다. (Day 1 에서 GPIO7 · 8 이 active-low consumer="spi0 CS0" 로 보였던 것 — SPI 의 CS 는 Low 에서 동작한다.)

2-5. drive — 출력 회로의 종류

gpiod_line_settings_set_drive() 1 을 쓰면 0 을 쓰면 쓰는 곳
PUSH_PULL (기본) 3.3V 로 밀어 냄 0V 로 끌어 내림 LED · 부저 대부분
OPEN_DRAIN 손을 놓음 (떠 있음 → 외부 풀업이 끌어올림) 0V 여러 장치가 한 선을 공유 (I2C 원리, Day 4)
OPEN_SOURCE 3.3V 손을 놓음 드묾

핀 3개 × 약 5.9 mA ≈ 18 mA. Pi 전체 GPIO 의 합계 전류도 수십 mA 수준이라 LED 는 220Ω 이상으로 유지한다.

3-1. 왜 레지스터를 보는가

MCU 과목에서 digitalWrite() 아래에는 결국 레지스터 쓰기가 있었다. Pi 에서도 GPIO 는 SoC 안의 레지스터이고, libgpiod → 커널 드라이버도 마지막에는 같은 레지스터를 쓴다. 오늘은 그 레지스터를 사용자 프로그램에서 직접 써 보고, 왜 평소에는 그렇게 하지 않는지 확인한다.

3-2. BCM2711 GPIO 레지스터 지도

오프셋 이름 한 칸의 의미 쓰는 법
0x00 ~ 0x14 GPFSEL0 ~ 5 핀당 3비트 기능 (000 입력 · 001 출력 · 그 외 ALT) 읽기-수정-쓰기
0x1C GPSET0 핀 0~31, 1 을 쓴 비트만 High 1 만 쓰면 됨 (0 은 무시)
0x28 GPCLR0 핀 0~31, 1 을 쓴 비트만 Low 1 만 쓰면 됨
0x34 GPLEV0 핀 0~31 현재 레벨 읽기
0xE4 ~ 0xF0 GPIO_PUP_PDN_CNTRL0 ~ 3 핀당 2비트 pull (00 없음 · 01 up · 10 down) 읽기-수정-쓰기

/dev/gpiomem 은 GPIO 레지스터 4KB 만 보여 주는 장치 파일이다 (gpio 그룹이면 열 수 있음). 물리 메모리 전체인 /dev/mem 과 달리 다른 하드웨어를 망가뜨릴 위험이 없다. Linux & Shell Day 3 에서 배운 mmap() 으로 이 파일을 메모리처럼 붙인다.

3-3. GPIO17 의 비트 계산

레지스터 계산 GPIO17
GPFSEL 번호 = pin / 10, 시작 비트 = (pin % 10) × 3 GPFSEL1, 비트 21~23
GPSET0 · GPCLR0 · GPLEV0 비트 = pin 비트 17 → 1u << 17
PUP_PDN 번호 = pin / 16, 시작 비트 = (pin % 16) × 2 CNTRL1, 비트 2~3 
/* GPIO17 을 출력으로: GPFSEL1 의 비트 21~23 을 001 로 */
v  = gpio[1];
v &= ~(7u << 21);        /* 3비트 지우기  (7 = 0b111)  */
v |=  (1u << 21);        /* 001 = 출력                 */
gpio[1] = v;

gpio[0x1C / 4] = 1u << 17;   /* GPSET0: High — 다른 핀은 건드리지 않는다 */
gpio[0x28 / 4] = 1u << 17;   /* GPCLR0: Low                             */

GPSET/GPCLR 이 따로 있는 이유: "17번만 High" 를 쓰기 한 번으로 끝내기 위해서다. 출력 레지스터 하나를 읽고-고치고-쓰면 그 사이에 다른 코드가 같은 레지스터를 바꿨을 때 그 변경을 덮어쓴다 (경쟁 조건, Linux & Shell Day 3).

3-4. 예제 03 — 레지스터만으로 버튼 → LED

src/03_gpiomem.c — 버튼 B 는 아직 쓰지 않는다.

/* 03_gpiomem.c — /dev/gpiomem 을 mmap 해 GPIO 레지스터를 직접 쓰기 (비교용 시연)
 * 배선: GPIO17 LED, GPIO5 버튼(→GND)
 * 동작: 버튼을 누르는 동안 LED 켜기 — libgpiod 없이 레지스터만으로
 *
 * ⚠ 수업 비교용 코드다. 커널이 모르는 사이에 핀 기능을 바꾸므로
 *   같은 핀을 libgpiod 로 쓰는 프로그램과 동시에 돌리면 안 된다.
 */
#include <fcntl.h>
#include <signal.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/mman.h>
#include <time.h>
#include <unistd.h>

/* BCM2711 GPIO 레지스터 (32비트 워드 인덱스 = 바이트 오프셋 / 4) */
#define GPFSEL0   0                  /* 0x00 : GPIO 0~9   기능 선택, 핀당 3비트 */
#define GPSET0    (0x1C / 4)         /* 0x1C : 1 을 쓴 비트의 핀 = High */
#define GPCLR0    (0x28 / 4)         /* 0x28 : 1 을 쓴 비트의 핀 = Low  */
#define GPLEV0    (0x34 / 4)         /* 0x34 : 핀 현재 레벨 읽기 */
#define PUPPDN0   (0xE4 / 4)         /* 0xE4 : GPIO 0~15 pull 설정, 핀당 2비트 (BCM2711) */

#define LED 17
#define BTN 5

static volatile sig_atomic_t g_stop = 0;
static void on_sigint(int s) { (void)s; g_stop = 1; }

static void set_function(volatile uint32_t *gpio, int pin, int fsel)
{
    int reg = GPFSEL0 + pin / 10;          /* GPIO17 → GPFSEL1 */
    int shift = (pin % 10) * 3;            /* GPIO17 → 비트 21~23 */
    uint32_t v = gpio[reg];
    v &= ~(7u << shift);                   /* 3비트 지우기 */
    v |= ((uint32_t)fsel << shift);        /* 000 = 입력, 001 = 출력 */
    gpio[reg] = v;
}

int main(void)
{
    signal(SIGINT, on_sigint);

    int fd = open("/dev/gpiomem", O_RDWR | O_SYNC);
    if (fd < 0) { perror("open /dev/gpiomem"); return EXIT_FAILURE; }

    volatile uint32_t *gpio = mmap(NULL, 4096, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    close(fd);                              /* 매핑은 fd 를 닫아도 남는다 */
    if (gpio == MAP_FAILED) { perror("mmap"); return EXIT_FAILURE; }

    set_function(gpio, LED, 1);             /* GPIO17 = 출력 */
    set_function(gpio, BTN, 0);             /* GPIO5  = 입력 */
    uint32_t p = gpio[PUPPDN0];             /* GPIO5 pull-up: 비트 10~11 = 01 */
    p &= ~(3u << (BTN * 2));
    p |=  (1u << (BTN * 2));
    gpio[PUPPDN0] = p;

    printf("버튼을 누르는 동안 LED 켜짐 (Ctrl+C 로 종료)\n");
    while (!g_stop) {
        int pressed = !(gpio[GPLEV0] & (1u << BTN));     /* pull-up → 누르면 0 */
        if (pressed) gpio[GPSET0] = 1u << LED;            /* 읽기-수정-쓰기 불필요 */
        else         gpio[GPCLR0] = 1u << LED;
        struct timespec ts = { 0, 1000000L };             /* 1 ms 폴링 */
        nanosleep(&ts, NULL);
    }

    gpio[GPCLR0] = 1u << LED;
    set_function(gpio, LED, 0);             /* 원래대로 입력으로 돌려 두기 */
    munmap((void *)gpio, 4096);
    printf("\n종료\n");
    return EXIT_SUCCESS;
}

3-5. 예제 04 — 토글 속도 비교

src/04_speed.c

/* 04_speed.c — 같은 핀을 libgpiod 와 mmap 레지스터로 N 번 토글해 속도 비교
 * 배선: GPIO17 LED (깜빡임이 너무 빨라 눈에는 반쯤 켜진 것처럼 보인다)
 * 실행: ./04_speed            ./04_speed 2000000
 */
#include "gpio_common.h"
#include <fcntl.h>
#include <stdint.h>
#include <stdlib.h>
#include <sys/mman.h>
#include <unistd.h>

#define LED 17

int main(int argc, char **argv)
{
    long n = (argc > 1) ? atol(argv[1]) : 1000000;
    unsigned int off = LED;

    /* ① libgpiod: 토글마다 ioctl 시스템 콜 1번 */
    struct gpiod_line_request *req = gpio_request_output("speed", &off, 1);
    if (!req) return EXIT_FAILURE;
    long long t0 = now_ns();
    for (long i = 0; i < n; i++)
        gpiod_line_request_set_value(req, off, (i & 1) ? GPIOD_LINE_VALUE_ACTIVE
                                                       : GPIOD_LINE_VALUE_INACTIVE);
    long long t1 = now_ns();
    gpiod_line_request_release(req);        /* mmap 실험 전에 반납 */

    /* ② mmap: 토글마다 메모리 쓰기 1번 (커널을 거치지 않음) */
    int fd = open("/dev/gpiomem", O_RDWR | O_SYNC);
    if (fd < 0) { perror("open /dev/gpiomem"); return EXIT_FAILURE; }
    volatile uint32_t *g = mmap(NULL, 4096, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    close(fd);
    if (g == MAP_FAILED) { perror("mmap"); return EXIT_FAILURE; }
    g[1] = (g[1] & ~(7u << 21)) | (1u << 21);           /* GPFSEL1: GPIO17 = 출력 */
    long long t2 = now_ns();
    for (long i = 0; i < n; i++) {
        if (i & 1) g[0x1C / 4] = 1u << LED;             /* GPSET0 */
        else       g[0x28 / 4] = 1u << LED;             /* GPCLR0 */
    }
    long long t3 = now_ns();
    g[0x28 / 4] = 1u << LED;
    g[1] &= ~(7u << 21);                                /* 다시 입력으로 */
    munmap((void *)g, 4096);

    double a = (t1 - t0) / (double)n, b = (t3 - t2) / (double)n;
    printf("토글 %ld 번\n", n);
    printf("  libgpiod : %8.1f ns/회  (%6.2f MHz)\n", a, 1000.0 / a);
    printf("  mmap     : %8.1f ns/회  (%6.2f MHz)\n", b, 1000.0 / b);
    printf("  mmap 이 약 %.0f 배 빠름\n", a / b);
    return EXIT_SUCCESS;
}

3-6. 그럼 왜 평소에는 mmap 을 쓰지 않나

문제 설명
소유권 없음 커널은 누가 핀을 쓰는지 모른다 → busy 보호가 없어 다른 프로그램 · 드라이버와 충돌
이식성 없음 레지스터 주소 · 배치는 SoC 마다 다르다 (Pi 5 는 RP1 칩이라 완전히 다름)
인터럽트 없음 엣지를 기다리려면 계속 읽어야 한다 (폴링)
정리 책임 프로그램이 죽으면 핀이 출력으로 남는다 — 커널이 되돌려 주지 않는다

결론: 1 MHz 이상의 아주 빠른 비트뱅잉이 꼭 필요할 때만 쓰고, 나머지는 libgpiod 를 쓴다. 정확한 파형이 필요하면 하드웨어 PWM · SPI (Day 3 · 4) 를 쓴다.

src/05_button_poll.c

/* 05_button_poll.c — 폴링으로 버튼 읽기 + 소프트웨어 디바운스
 * 배선: GPIO5 버튼(→GND), 내부 pull-up 사용 (외부 저항 없음)
 * 실행: ./05_button_poll          디바운스 켜고 (기본)
 *       ./05_button_poll -n       디바운스 없이 → 바운싱이 그대로 보인다
 *       ./05_button_poll -n 6     GPIO6 (기울기 센서) 를 읽기
 */
#include "gpio_common.h"
#include <stdlib.h>

#define POLL_MS     1      /* 1 ms 마다 읽기 */
#define STABLE_CNT  20     /* 20번(=20 ms) 연속 같은 값이어야 인정 */

int main(int argc, char **argv)
{
    int debounce = 1;
    unsigned int line = 5;
    for (int i = 1; i < argc; i++) {
        if (strcmp(argv[i], "-n") == 0) debounce = 0;
        else line = (unsigned int)atoi(argv[i]);
    }
    gpio_install_sigint();

    struct gpiod_line_request *req = gpio_request_input("button_poll", &line, 1,
                                                        GPIOD_LINE_EDGE_NONE, 0);
    if (!req) return EXIT_FAILURE;

    int stable = gpiod_line_request_get_value(req, line);   /* 확정된 값 */
    int last = stable, same = 0, presses = 0;
    printf("GPIO%u 폴링 중 (%s) — Ctrl+C 로 종료\n", line,
           debounce ? "디바운스 20 ms" : "디바운스 없음");

    while (!g_stop) {
        int v = gpiod_line_request_get_value(req, line);    /* 1 = 안 누름, 0 = 누름 */
        if (v < 0) { perror("get_value"); break; }

        if (!debounce) {
            if (v != stable) {                               /* 바뀌는 즉시 인정 */
                if (v == 0) printf("  눌림 #%d\n", ++presses);
                stable = v;
            }
        } else {
            same = (v == last) ? same + 1 : 0;               /* 같은 값이 몇 번 이어졌나 */
            last = v;
            if (same >= STABLE_CNT && v != stable) {         /* 20 ms 동안 안정 → 확정 */
                stable = v;
                if (v == 0) printf("  눌림 #%d\n", ++presses);
            }
        }
        sleep_ms(POLL_MS);
    }

    gpiod_line_request_release(req);
    printf("\n총 %d 번 눌림\n", presses);
    return EXIT_SUCCESS;
}

5-3. 예제 06 — 이벤트로 짧게/길게 누름 구분

src/06_edge_poll.c

/* 06_edge_poll.c — 커널 엣지 이벤트를 poll() 로 기다리기
 * 배선: GPIO5 버튼(→GND), GPIO17 LED
 * 동작: 짧게 누르면 LED 토글, 0.5초 이상 길게 누르면 "길게" 출력
 *       기다리는 동안 CPU 를 쓰지 않는다 (top 으로 05 와 비교)
 * 실행: ./06_edge_poll        ./06_edge_poll 0   (디바운스 끄기 → 바운싱 관찰)
 */
#include "gpio_common.h"
#include <poll.h>
#include <stdlib.h>

#define BTN 5
#define LED 17
#define LONG_MS 500

int main(int argc, char **argv)
{
    unsigned long deb_us = (argc > 1) ? strtoul(argv[1], NULL, 10) * 1000 : 10000;
    unsigned int btn = BTN, led = LED;
    gpio_install_sigint();

    struct gpiod_line_request *in  = gpio_request_input("edge_poll", &btn, 1,
                                                        GPIOD_LINE_EDGE_BOTH, deb_us);
    struct gpiod_line_request *out = gpio_request_output("edge_poll", &led, 1);
    if (!in || !out) return EXIT_FAILURE;

    struct gpiod_edge_event_buffer *buf = gpiod_edge_event_buffer_new(16);
    struct pollfd pfd = { .fd = gpiod_line_request_get_fd(in), .events = POLLIN };
    long long t_down = 0;
    int led_on = 0;

    printf("버튼 이벤트 대기 (디바운스 %lu ms) — Ctrl+C 로 종료\n", deb_us / 1000);
    while (!g_stop) {
        int r = poll(&pfd, 1, -1);                  /* 이벤트가 올 때까지 잠든다 */
        if (r < 0) { if (errno == EINTR) continue; perror("poll"); break; }

        int n = gpiod_line_request_read_edge_events(in, buf, 16);
        for (int i = 0; i < n; i++) {
            struct gpiod_edge_event *ev = gpiod_edge_event_buffer_get_event(buf, i);
            long long ts = (long long)gpiod_edge_event_get_timestamp_ns(ev);
            int falling = gpiod_edge_event_get_event_type(ev) == GPIOD_EDGE_EVENT_FALLING_EDGE;

            printf("  %lld.%09lld  %s\n", ts / 1000000000LL, ts % 1000000000LL,
                   falling ? "falling (누름)" : "rising  (뗌)");

            if (falling) {
                t_down = ts;
            } else if (t_down) {
                long ms = (long)((ts - t_down) / 1000000);
                if (ms >= LONG_MS) {
                    printf("    → 길게 %ld ms\n", ms);
                } else {
                    led_on = !led_on;
                    gpiod_line_request_set_value(out, led,
                        led_on ? GPIOD_LINE_VALUE_ACTIVE : GPIOD_LINE_VALUE_INACTIVE);
                    printf("    → 짧게 %ld ms, LED %s\n", ms, led_on ? "ON" : "OFF");
                }
                t_down = 0;
            }
        }
    }

    gpiod_line_request_set_value(out, led, GPIOD_LINE_VALUE_INACTIVE);
    gpiod_edge_event_buffer_free(buf);
    gpiod_line_request_release(in);
    gpiod_line_request_release(out);
    printf("\n종료\n");
    return EXIT_SUCCESS;
}

6-3. timerfd — 타이머도 fd 로

timerfd_create() 로 만든 fd 는 정해진 시간이 되면 읽을 수 있는 상태가 된다. 읽으면 그동안 몇 번 만료됐는지(uint64)가 나온다. sleep() 과 달리 프로그램이 잠들지 않고, 다른 fd 와 함께 epoll 로 기다릴 수 있다.

함수 하는 일
timerfd_create(CLOCK_MONOTONIC, 0) 타이머 fd 생성
timerfd_settime(fd, 0, &its, NULL) it_value = 첫 만료, it_interval = 주기
read(fd, &exp, 8) 만료 횟수 읽기 (읽어야 다시 대기 상태)

6-4. 예제 07 — 이벤트 루프

src/07_evloop.c

/* 07_evloop.c — epoll 하나로 여러 입력을 동시에 기다리는 이벤트 루프
 *   fd ① 버튼 A  (GPIO5, falling)       → 누른 횟수 세기
 *   fd ② PIR    (GPIO6, both)          → 움직임 시작/끝 로그
 *   fd ③ timerfd (0.5초 주기)           → LED(GPIO17) 깜빡임
 *   fd ④ stdin  ("s" Enter = 통계, "q" Enter = 종료)
 * 배선: GPIO5 버튼(→GND), GPIO6 PIR OUT (PIR VCC 는 3V3), GPIO17 LED
 */
#include "gpio_common.h"
#include <stdint.h>
#include <stdlib.h>
#include <sys/epoll.h>
#include <sys/timerfd.h>
#include <unistd.h>

enum { TAG_BTN = 1, TAG_PIR, TAG_TIMER, TAG_STDIN };

static int add_fd(int ep, int fd, int tag)
{
    struct epoll_event e = { .events = EPOLLIN, .data.u32 = (uint32_t)tag };
    return epoll_ctl(ep, EPOLL_CTL_ADD, fd, &e);
}

int main(void)
{
    unsigned int btn = 5, pir = 6, led = 17;
    gpio_install_sigint();

    struct gpiod_line_request *rb = gpio_request_input("evloop", &btn, 1,
                                                       GPIOD_LINE_EDGE_FALLING, 10000);
    /* PIR 은 스스로 0/1 을 내보내므로 bias 를 끄고 요청한다 */
    struct gpiod_line_settings *s = gpiod_line_settings_new();
    gpiod_line_settings_set_direction(s, GPIOD_LINE_DIRECTION_INPUT);
    gpiod_line_settings_set_bias(s, GPIOD_LINE_BIAS_DISABLED);
    gpiod_line_settings_set_edge_detection(s, GPIOD_LINE_EDGE_BOTH);
    struct gpiod_line_request *rp = gpio_request("evloop", &pir, 1, s);
    gpiod_line_settings_free(s);
    struct gpiod_line_request *rl = gpio_request_output("evloop", &led, 1);
    if (!rb || !rp || !rl) return EXIT_FAILURE;

    int tfd = timerfd_create(CLOCK_MONOTONIC, 0);
    struct itimerspec its = { .it_interval = { 0, 500000000L }, .it_value = { 0, 500000000L } };
    timerfd_settime(tfd, 0, &its, NULL);

    int ep = epoll_create1(0);
    add_fd(ep, gpiod_line_request_get_fd(rb), TAG_BTN);
    add_fd(ep, gpiod_line_request_get_fd(rp), TAG_PIR);
    add_fd(ep, tfd, TAG_TIMER);
    add_fd(ep, STDIN_FILENO, TAG_STDIN);

    struct gpiod_edge_event_buffer *buf = gpiod_edge_event_buffer_new(8);
    long presses = 0, motions = 0, ticks = 0;
    int led_on = 0;
    printf("이벤트 루프 시작 — s=통계, q=종료 (Enter)\n");

    while (!g_stop) {
        struct epoll_event evs[4];
        int n = epoll_wait(ep, evs, 4, -1);            /* 준비된 fd 목록만 돌려준다 */
        if (n < 0) { if (errno == EINTR) continue; perror("epoll_wait"); break; }

        for (int i = 0; i < n; i++) {
            switch (evs[i].data.u32) {
            case TAG_BTN: {
                int k = gpiod_line_request_read_edge_events(rb, buf, 8);
                presses += k;
                printf("[버튼] %ld 번째\n", presses);
                break;
            }
            case TAG_PIR: {
                int k = gpiod_line_request_read_edge_events(rp, buf, 8);
                for (int j = 0; j < k; j++) {
                    struct gpiod_edge_event *ev = gpiod_edge_event_buffer_get_event(buf, j);
                    int rise = gpiod_edge_event_get_event_type(ev) == GPIOD_EDGE_EVENT_RISING_EDGE;
                    if (rise) motions++;
                    printf("[PIR ] %s\n", rise ? "움직임 감지" : "조용해짐");
                }
                break;
            }
            case TAG_TIMER: {
                uint64_t exp;
                if (read(tfd, &exp, sizeof exp) == sizeof exp) ticks += (long)exp;
                led_on = !led_on;
                gpiod_line_request_set_value(rl, led, led_on ? GPIOD_LINE_VALUE_ACTIVE
                                                             : GPIOD_LINE_VALUE_INACTIVE);
                break;
            }
            case TAG_STDIN: {
                char line[32];
                if (!fgets(line, sizeof line, stdin) || line[0] == 'q') g_stop = 1;
                else if (line[0] == 's')
                    printf("[통계] 버튼 %ld · 움직임 %ld · 타이머 %ld\n", presses, motions, ticks);
                break;
            }
            }
        }
    }

    gpiod_line_request_set_value(rl, led, GPIOD_LINE_VALUE_INACTIVE);
    gpiod_edge_event_buffer_free(buf);
    close(ep); close(tfd);
    gpiod_line_request_release(rb);
    gpiod_line_request_release(rp);
    gpiod_line_request_release(rl);
    printf("종료\n");
    return EXIT_SUCCESS;
}

8-1. 완성본 — react.c

src/react.c — 조별로 짠 코드와 비교한다.

/* react.c — 미니 프로젝트: 반응속도 게임 (MCU Day 1 과 같은 규칙)
 *
 *   1) "준비" 후 2~5초 무작위로 기다린다
 *   2) LED(GPIO17) 가 켜지는 순간부터 버튼을 누를 때까지의 시간을 잰다
 *      — 측정은 커널 엣지 타임스탬프 − LED 를 켠 시각 (둘 다 CLOCK_MONOTONIC)
 *   3) LED 가 켜지기 전에 누르면 부정 출발 → 부저 길게
 *   4) 라운드가 끝나면 평균 · 최고 기록을 출력하고 react_log.csv 에 추가
 *
 * 배선: GPIO17 LED · GPIO5 버튼 A · GPIO6 버튼 B(2인용) · GPIO13 액티브 부저
 * 실행: ./react            (1인, 5라운드)
 *       ./react 10         (1인, 10라운드)
 *       ./react -2         (2인 대전: 먼저 누른 사람 승, 부정 출발은 패)
 */
#include "gpio_common.h"
#include <poll.h>
#include <stdlib.h>

#define LED 17
#define BTN_A 5
#define BTN_B 6
#define BUZ 13

static struct gpiod_line_request *g_out, *g_in;
static struct gpiod_edge_event_buffer *g_buf;

static void set_line(unsigned int off, int on)
{
    gpiod_line_request_set_value(g_out, off, on ? GPIOD_LINE_VALUE_ACTIVE
                                                : GPIOD_LINE_VALUE_INACTIVE);
}

static void beep(int ms) { set_line(BUZ, 1); sleep_ms(ms); set_line(BUZ, 0); }

/* 쌓여 있던 이벤트 버리기 (이전 라운드에서 늦게 뗀 것 등) */
static void drain(void)
{
    struct pollfd p = { .fd = gpiod_line_request_get_fd(g_in), .events = POLLIN };
    while (poll(&p, 1, 0) > 0)
        gpiod_line_request_read_edge_events(g_in, g_buf, 16);
}

/* timeout_ms 동안 누름(falling) 을 기다린다.
 * 반환: 1 = 눌림 (*ts, *who 채움), 0 = 시간 초과, -1 = Ctrl+C */
static int wait_press(int timeout_ms, long long *ts, unsigned int *who)
{
    struct pollfd p = { .fd = gpiod_line_request_get_fd(g_in), .events = POLLIN };
    long long deadline = now_ns() + (long long)timeout_ms * 1000000LL;
    for (;;) {
        if (g_stop) return -1;
        int left = timeout_ms < 0 ? -1 : (int)((deadline - now_ns()) / 1000000LL);
        if (timeout_ms >= 0 && left <= 0) return 0;
        int r = poll(&p, 1, left);
        if (r < 0) { if (errno == EINTR) continue; return -1; }
        if (r == 0) return 0;
        int n = gpiod_line_request_read_edge_events(g_in, g_buf, 16);
        for (int i = 0; i < n; i++) {
            struct gpiod_edge_event *ev = gpiod_edge_event_buffer_get_event(g_buf, i);
            if (gpiod_edge_event_get_event_type(ev) != GPIOD_EDGE_EVENT_FALLING_EDGE)
                continue;
            *ts = (long long)gpiod_edge_event_get_timestamp_ns(ev);
            *who = gpiod_edge_event_get_line_offset(ev);
            return 1;
        }
    }
}

int main(int argc, char **argv)
{
    int duo = 0, rounds = 5;
    for (int i = 1; i < argc; i++) {
        if (strcmp(argv[i], "-2") == 0) duo = 1;
        else rounds = atoi(argv[i]) > 0 ? atoi(argv[i]) : rounds;
    }
    gpio_install_sigint();
    srand((unsigned)now_ns());

    unsigned int outs[2] = { LED, BUZ };
    unsigned int ins[2] = { BTN_A, BTN_B };
    g_out = gpio_request_output("react", outs, 2);
    g_in  = gpio_request_input("react", ins, duo ? 2 : 1, GPIOD_LINE_EDGE_FALLING, 5000);
    g_buf = gpiod_edge_event_buffer_new(16);
    if (!g_out || !g_in || !g_buf) return EXIT_FAILURE;

    long best = 0, sum = 0; int ok = 0, fouls = 0, winA = 0, winB = 0;
    printf("반응속도 게임 — %s, %d 라운드 (Ctrl+C 로 중단)\n", duo ? "2인 대전" : "1인", rounds);

    for (int r = 1; r <= rounds && !g_stop; r++) {
        long long ts; unsigned int who;
        printf("\n[%d/%d] 준비...\n", r, rounds);
        drain();

        int delay = 2000 + rand() % 3000;                 /* 2~5초 */
        int pr = wait_press(delay, &ts, &who);
        if (pr < 0) break;
        if (pr == 1) {                                    /* LED 가 켜지기 전에 누름 */
            fouls++;
            printf("  부정 출발! (%s)\n", who == BTN_A ? "A" : "B");
            if (duo) { if (who == BTN_A) winB++; else winA++; }
            beep(800);
            continue;
        }

        set_line(LED, 1);
        long long t_on = now_ns();                        /* LED 를 켠 시각 */
        pr = wait_press(3000, &ts, &who);
        set_line(LED, 0);
        if (pr < 0) break;
        if (pr == 0) { printf("  시간 초과 (3초)\n"); continue; }

        long ms = (long)((ts - t_on) / 1000000LL);         /* 커널 타임스탬프 기준 */
        printf("  %s%ld ms\n", duo ? (who == BTN_A ? "A 승 · " : "B 승 · ") : "", ms);
        if (duo) { if (who == BTN_A) winA++; else winB++; }
        sum += ms; ok++;
        if (!best || ms < best) best = ms;
        beep(100);
    }

    set_line(LED, 0); set_line(BUZ, 0);
    printf("\n== 결과 ==\n");
    if (ok) printf("성공 %d 회 · 평균 %ld ms · 최고 %ld ms · 부정 출발 %d 회\n",
                   ok, sum / ok, best, fouls);
    if (duo) printf("A %d : %d B\n", winA, winB);

    FILE *fp = fopen("react_log.csv", "a");
    if (fp && ok) {
        time_t t = time(NULL);
        char when[32];
        strftime(when, sizeof when, "%Y-%m-%d %H:%M:%S", localtime(&t));
        fprintf(fp, "%s,%s,%d,%ld,%ld,%d\n", when, duo ? "duo" : "solo", ok, sum / ok, best, fouls);
    }
    if (fp) fclose(fp);

    gpiod_edge_event_buffer_free(g_buf);
    gpiod_line_request_release(g_in);
    gpiod_line_request_release(g_out);
    return EXIT_SUCCESS;
}

make react
./react

8-3. 2인 대전

버튼 B 를 다시 GPIO6 에 꽂고:

./react -2

[1/5] 준비...
  B 승 · 187 ms
[2/5] 준비...
  부정 출발! (A)        ← A 가 먼저 눌러서 B 가 1점
...
A 2 : 3 B

두 버튼을 한 요청에 offsets {5, 6} 으로 묶었기 때문에 이벤트는 한 큐에 시간 순서대로 들어온다. 먼저 들어온 falling 의 line_offset 이 승자다.

8-4. 부하를 걸고 다시 해 보기

다른 창에서 CPU 부하를 걸고 게임을 한다.

yes > /dev/null &  yes > /dev/null &  yes > /dev/null &  yes > /dev/null &
./react
killall yes                          # 끝나면 반드시 정리

반응 시간이 거의 변하지 않는다 — 커널 타임스탬프를 쓰기 때문이다. 만약 wait_press() 에서 *ts = now_ns() 로 바꿔 빌드하면 부하 중에 기록이 들쭉날쭉해진다. (🎬 D2-8 에서 미리 볼 수 있다.)

  • [ ] 최고 기록을 best.txt 에 저장해 두고, 깨면 부저를 세 번 울린다.
  • [ ] 07 의 epoll 구조로 바꿔, 게임 중에도 키보드 q 로 중단할 수 있게 한다.
  • [ ] *ts = now_ns() 버전과 커널 타임스탬프 버전을 부하 중에 10번씩 측정해 평균 · 표준편차를 비교한다.

시드를 현재 나노초로 주면 실행마다 지연이 달라집니다. rand() % 3000 은 0~2999 라서 2000 을 더하면 2~5초입니다. ts(커널 타임스탬프)와 t_on 이 같은 CLOCK_MONOTONIC 이라 바로 뺄 수 있고, "a" 모드는 파일 끝에 덧붙입니다.

✏️ 괄호 넣기 8-2

  1. 2인 모드에서 두 버튼은 요청한다.
  2. 두 사람 중 누가 먼저 눌렀는지는 이벤트의 line 으로 안다.
  3. 부하를 걸어도 반응 시간이 정확한 이유는 커널이 순간에 시각을 찍기 때문이다.
  4. CSV 파일에 기록을 계속 쌓으려면 fopen 모드를 "" 로 연다.
🔒 정답 보기 8-2
  1. 한 요청에 함께
  2. offset
  3. 인터럽트
  4. a

한 요청에 묶은 라인들의 이벤트는 하나의 큐에 시간 순서대로 쌓입니다. 커널 타임스탬프는 인터럽트를 처리하는 순간 기록되므로 사용자 프로세스가 늦게 깨어나도 값이 변하지 않습니다(🎬 D2-8).
