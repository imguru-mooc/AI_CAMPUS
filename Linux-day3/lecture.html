🎬 D3-1 주소 공간 지도 — 변수를 눌러 영역 찾기, ASLR 로 시작 주소 섞기

1-1. Day 2 복습과 오늘의 질문

Day 2 에서는 CPU 를 누가, 얼마나, 얼마나 늦게 받는지 쟀다. 오늘은 메모리다. 프로세스마다 0 부터 수십 TB 까지의 가상 주소 공간을 혼자 쓰는 것처럼 보이지만, 실제 RAM(물리 메모리)은 VM 전체에 4GB 뿐이다. 그 사이를 페이지 테이블과 page fault 가 이어 준다. 오늘은 그 연결이 언제 일어나는지를 page fault 수로 증명한다.

실습 전 점검: Day 2 1-0 의 거북이 아이콘 문제가 해결된 VM 에서 진행한다. 오늘 측정은 page fault 수가 중심이라 시간 지연의 영향은 적지만, 시간(ms) 열은 흔들릴 수 있다.

1-2. 주소 공간의 지도

영역 무엇이 들어가나 권한 (maps)
text 기계어 코드 (main 등) r-xp
rodata 문자열 상수 "hello" r--p
data 초기값 있는 전역 · static 변수 rw-p
bss 초기값 없는 전역 · static 변수 (0 으로 시작) rw-p
heap 작은 malloc — brk 로 위로 늘어남 rw-p [heap]
mmap 영역 큰 malloc, 공유 라이브러리, 파일 매핑 rw-p / r-xp
stack 지역 변수, 함수 호출 기록 — 아래로 자람 rw-p [stack]

src/01_layout.c

/* 01_layout.c — 변수 종류별 주소와 /proc/self/maps 대조
 * 빌드: make 01_layout
 * 실행: ./01_layout             (두 번 실행해 주소가 바뀌는지 — ASLR)
 *       setarch -R ./01_layout  (ASLR 끄고 실행 — 매번 같은 주소)
 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int g_init = 42;              /* data : 초기값 있는 전역 */
int g_zero;                   /* bss  : 초기값 없는 전역 */
const char *g_msg = "hello";  /* "hello" 는 rodata */

static void show_maps(void)
{
    FILE *fp = fopen("/proc/self/maps", "r");
    char line[512];
    if (fp == NULL)
        return;
    printf("\n--- /proc/self/maps (주요 줄) ---\n");
    while (fgets(line, sizeof(line), fp) != NULL)
        if (strstr(line, "01_layout") || strstr(line, "[heap]") || strstr(line, "[stack]"))
            printf("%s", line);
    fclose(fp);
}

int main(void)
{
    static int s_local = 7;   /* data : 함수 안 static */
    int local = 1;            /* stack */
    int *small = malloc(64);           /* heap (brk 영역) */
    char *big = malloc(1024 * 1024);   /* 1MB → mmap 영역 */

    printf("%-26s %p\n", "text   main()", (void *)main);
    printf("%-26s %p\n", "rodata \"hello\"", (void *)g_msg);
    printf("%-26s %p\n", "data   g_init", (void *)&g_init);
    printf("%-26s %p\n", "data   static s_local", (void *)&s_local);
    printf("%-26s %p\n", "bss    g_zero", (void *)&g_zero);
    printf("%-26s %p\n", "heap   malloc(64)", (void *)small);
    printf("%-26s %p\n", "mmap   malloc(1MB)", (void *)big);
    printf("%-26s %p\n", "stack  local", (void *)&local);
    show_maps();

    free(small);
    free(big);
    return 0;
}

ASLR(Address Space Layout Randomization) 은 실행할 때마다 영역의 시작 주소를 무작위로 옮겨, 공격자가 주소를 예측하지 못하게 한다. setarch -R 로 끄면 매번 0x5555555… 처럼 같은 주소가 나온다.

int g_init = 42;              /* data : 초기값 있는 전역 */
int g_zero;                   /* bss  : 초기값 없는 전역 */
    int local = 1;            /* stack */
    int *small = malloc(64);  /* heap (brk 영역) */

초기값이 있는 전역 변수는 실행 파일에 값이 저장되는 data, 초기값이 없는 전역은 0 으로 시작하는 bss 에 들어갑니다(파일에는 크기만 기록). 지역 변수는 stack, 작은 malloc 은 heap(brk 영역)입니다. /proc/self/maps 는 지금 이 프로세스의 주소 공간 지도입니다.

✏️ 괄호 넣기 1-2

  1. 초기값 없는 전역 변수가 들어가는 영역은 이다.
  2. /proc/self/maps 에서 기계어 코드 영역의 권한은 이다.
  3. 실행할 때마다 주소를 무작위로 바꾸는 기능은 이다.
  4. 스택은 주소가 자란다.
🔒 정답 보기 1-2
  1. bss
  2. r-xp
  3. ASLR
  4. 낮은 쪽으로

bss 는 0 으로 시작하는 전역·static 변수 영역입니다. 기계어 코드는 읽기·실행만 가능한 r-xp 로 매핑됩니다. 실행마다 시작 주소를 무작위로 바꾸는 것이 ASLR 이고 setarch -R 로 끌 수 있습니다. 스택은 높은 주소에서 시작해 낮은 주소 쪽으로 자랍니다(🎬 D3-1).

malloc 내부

10:00–10:50

학습 목표

  • strace 로 malloc 이 부르는 시스템 콜을 확인한다
  • mallopt 로 기준값을 바꿔 결과를 비교한다

🎬 D3-2 malloc 의 두 길 — 크기 슬라이더로 heap · mmap 선택과 free 동작 보기

2-1. malloc 은 시스템 콜이 아니다

malloc 은 glibc 의 함수다. 커널에서 메모리를 받아 오는 시스템 콜은 따로 있다.

방법 시스템 콜 언제 free 하면
heap 늘리기 brk 작은 요청 (기본 128KB 미만) glibc 가 보관해 다시 씀 — 커널에 바로 돌려주지 않음
독립 매핑 mmap 큰 요청 (128KB 이상) munmap 으로 즉시 커널에 반환

기준값은 M_MMAP_THRESHOLD(기본 128KB)이고 mallopt 로 바꿀 수 있다. glibc 는 처음 malloc 할 때 heap 을 132KB 정도 한 번에 늘려 두고, 작은 요청은 그 안에서 잘라 준다.

2-2. 예제 02 — 크기에 따라 길이 바뀐다

src/02_malloc_path.c

/* 02_malloc_path.c — malloc 이 brk(heap) 와 mmap 중 무엇을 쓰는지 확인
 * 빌드: make 02_malloc_path
 * 실행: ./02_malloc_path 64        (64KB)
 *       ./02_malloc_path 256       (256KB)
 *       strace -e trace=brk,mmap,munmap ./02_malloc_path 256
 *       ./02_malloc_path 256 512   (mallopt 로 기준을 512KB 로 올리면?)
 */
#include <stdio.h>
#include <stdlib.h>
#include <malloc.h>
#include <unistd.h>

int main(int argc, char *argv[])
{
    size_t kb = (argc > 1) ? strtoul(argv[1], NULL, 10) : 64;
    if (argc > 2)
        mallopt(M_MMAP_THRESHOLD, (int)strtoul(argv[2], NULL, 10) * 1024);

    char *heap_start = sbrk(0);                /* 프로그램 시작 시 heap 위치 */
    free(malloc(1));                           /* malloc 초기화(첫 132KB 준비)를 미리 끝낸다 */
    void *brk_before = sbrk(0);                /* 현재 heap 의 끝 */
    char *p = malloc(kb * 1024);
    void *brk_after = sbrk(0);
    if (p == NULL) {
        perror("malloc");
        return 1;
    }
    int in_heap = (p >= heap_start && p < (char *)brk_after);
    printf("malloc(%zu KB) = %p\n", kb, (void *)p);
    printf("heap 끝(brk): %p → %p  (%+ld KB)\n", brk_before, brk_after,
           ((char *)brk_after - (char *)brk_before) / 1024);
    printf("판정: %s\n", in_heap ? "heap (brk 영역)" : "mmap (독립된 영역 — free 하면 munmap 으로 바로 반환)");
    free(p);
    printf("free 후 brk: %p\n", sbrk(0));
    return 0;
}

✏️ 빈칸 채우기 2-1

큰 요청도 heap 에서 받도록 기준을 바꾸고, 현재 heap 끝을 확인합니다. 빈칸을 채우세요.

#include <>                         /* mallopt 선언 */
...
mallopt(, 512 * 1024);      /* 기준을 512KB 로 */
void *before = (0);                      /* 현재 heap 끝 */
char *p = malloc(256 * 1024);                   /* 이제 heap 에서 */
(p);
🔒 정답 보기 2-1 
#include <malloc.h>                         /* mallopt 선언 */
mallopt(M_MMAP_THRESHOLD, 512 * 1024);      /* 기준을 512KB 로 */
void *before = sbrk(0);                      /* 현재 heap 끝 */
free(p);

mallopt 는 <malloc.h> 에 선언되어 있습니다. M_MMAP_THRESHOLD 를 512KB 로 올리면 256KB 요청도 heap(brk)에서 받습니다. sbrk(0) 은 heap 을 늘리지 않고 현재 끝 주소만 돌려줍니다. 받은 메모리는 free 로 돌려줍니다.

✏️ 괄호 넣기 2-2

  1. 기본 설정에서 malloc 이 mmap 을 쓰기 시작하는 크기는 KB 이다.
  2. heap 의 끝을 옮기는 시스템 콜은 이다.
  3. mmap 으로 받은 큰 블록을 free 하면 glibc 는 으로 커널에 돌려준다.
  4. malloc 이 실제로 부르는 시스템 콜을 보려면 -e trace=brk,mmap 을 쓴다.
🔒 정답 보기 2-2
  1. 128
  2. brk
  3. munmap
  4. strace

기본 M_MMAP_THRESHOLD 는 128KB 입니다. heap 의 끝은 brk 시스템 콜로 옮기고, mmap 으로 받은 큰 블록은 free 하면 glibc 가 munmap 으로 즉시 커널에 반환합니다. strace -e trace=brk,mmap,munmap 으로 실제 호출을 볼 수 있습니다(🎬 D3-2).

Demand paging

학습 목표

🎬 D3-3 demand paging — malloc → 일부 쓰기 → 전체 쓰기를 페이지 격자로

3-1. 주소만 먼저, 페이지는 나중에

malloc(100MB) 는 주소 공간만 예약한다. 물리 페이지는 그 주소를 처음 읽거나 쓸 때 받는다.

CPU 가 주소 접근 → MMU 가 페이지 테이블 확인 → 매핑 없음 → page fault (커널 진입)
→ 커널이 빈 물리 페이지(4KB)를 0 으로 채워 연결 → 그 명령을 다시 실행
이름 의미
VmSize 예약한 가상 주소 전체 크기
VmRSS 지금 실제 RAM 에 올라와 있는 크기 (Resident Set Size)
minor fault 디스크 없이 처리 — 새 0 페이지, page cache 에 이미 있는 페이지
major fault 디스크에서 읽어 와야 함 — 훨씬 느림

3-2. 예제 03 — 숫자로 증명

src/memutil.h

/* memutil.h — Day 3 예제 공용 측정 함수 (헤더만 include 하면 된다) */
#ifndef MEMUTIL_H
#define MEMUTIL_H
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>
#include <sys/resource.h>

/* /proc/self/status 에서 key(예: "VmRSS") 값을 kB 로 */
static inline long status_kb(const char *key)
{
    FILE *fp = fopen("/proc/self/status", "r");
    char line[256];
    long v = -1;
    size_t n = strlen(key);
    if (fp == NULL)
        return -1;
    while (fgets(line, sizeof(line), fp) != NULL)
        if (strncmp(line, key, n) == 0 && line[n] == ':') {
            v = atol(line + n + 1);
            break;
        }
    fclose(fp);
    return v;
}

/* 지금까지의 minor page fault 수 (디스크 없이 처리된 fault) */
static inline long minflt(void)
{
    struct rusage ru;
    getrusage(RUSAGE_SELF, &ru);
    return ru.ru_minflt;
}

static inline long majflt(void)
{
    struct rusage ru;
    getrusage(RUSAGE_SELF, &ru);
    return ru.ru_majflt;
}

static inline double now_ms(void)
{
    struct timespec t;
    clock_gettime(CLOCK_MONOTONIC, &t);
    return t.tv_sec * 1000.0 + t.tv_nsec / 1e6;
}
#endif

src/03_demand.c

/* 03_demand.c — malloc 은 주소만 예약, 실제 페이지는 처음 만질 때 할당 (demand paging)
 * 빌드: make 03_demand
 * 실행: ./03_demand
 */
#include "memutil.h"
#include <unistd.h>

#define MB (1024 * 1024)

static void report(const char *step, long f0)
{
    printf("%-34s VmSize %7ld kB  VmRSS %7ld kB  minflt %+7ld\n",
           step, status_kb("VmSize"), status_kb("VmRSS"), minflt() - f0);
}

int main(void)
{
    long page = sysconf(_SC_PAGESIZE);
    long f0 = minflt();
    report("시작", f0);

    char *p = malloc(100 * MB);
    if (p == NULL)
        return 1;
    report("malloc(100MB) 직후", f0);

    for (long i = 0; i < 50 * MB; i += page)   /* 앞 50MB 를 4KB 마다 1바이트씩 */
        p[i] = 1;
    report("앞 50MB, 페이지마다 1바이트 쓰기", f0);

    memset(p, 2, 100 * MB);                     /* 전체 쓰기 */
    report("memset 100MB", f0);

    printf("페이지 크기 = %ld 바이트, 50MB / 4KB = %ld 페이지\n", page, 50L * MB / page);
    free(p);
    return 0;
}

🎬 D3-4 mmap 과 page cache — 두 프로세스가 같은 파일을 매핑하고 쓸 때의 차이

4-1. 파일을 배열처럼

mmap 은 파일(또는 익명 메모리)을 주소 공간에 연결한다. read/write 로 복사하지 않고 포인터로 바로 읽고 쓴다. 실제 데이터는 커널의 page cache 에 있고, 프로세스는 그 페이지를 자기 페이지 테이블로 가리킨다.

플래그 쓰면 용도
MAP_SHARED page cache 를 직접 고침 → 파일과 다른 프로세스에 보인다 파일 수정, 프로세스 간 공유
MAP_PRIVATE 쓰는 순간 내 복사본 생성(COW) → 파일은 그대로 실행 파일 로딩, 읽기 위주
MAP_ANONYMOUS 파일 없이 0 으로 시작 큰 malloc, 공유 메모리

msync(p, len, MS_SYNC) 는 고친 내용을 디스크에 즉시 쓰게 한다. 매핑은 close(fd) 를 해도 유지되며 munmap 으로 끝낸다.

4-2. 예제 04 — 대문자로 바꾸기

src/04_mmap_upper.c

/* 04_mmap_upper.c — 파일을 메모리에 매핑해 대문자로 바꾸기
 * 빌드: make 04_mmap_upper
 * 실행: echo "hello mmap world" > test.txt
 *       ./04_mmap_upper test.txt           (MAP_SHARED → 파일이 바뀐다)
 *       ./04_mmap_upper test.txt private   (MAP_PRIVATE → 내 복사본만 바뀐다)
 *       cat test.txt
 */
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/mman.h>
#include <sys/stat.h>

int main(int argc, char *argv[])
{
    if (argc < 2) {
        fprintf(stderr, "사용법: %s 파일 [private]\n", argv[0]);
        return 1;
    }
    int priv = (argc > 2 && strcmp(argv[2], "private") == 0);
    int fd = open(argv[1], O_RDWR);
    if (fd == -1) {
        perror("open");
        return 1;
    }
    struct stat st;
    fstat(fd, &st);
    if (st.st_size == 0) {
        fprintf(stderr, "빈 파일은 매핑할 수 없습니다\n");
        return 1;
    }
    char *p = mmap(NULL, st.st_size, PROT_READ | PROT_WRITE,
                   priv ? MAP_PRIVATE : MAP_SHARED, fd, 0);
    if (p == MAP_FAILED) {
        perror("mmap");
        return 1;
    }
    close(fd);                                  /* 매핑은 fd 를 닫아도 유지된다 */

    for (off_t i = 0; i < st.st_size; i++)      /* 배열처럼 바로 고친다 */
        p[i] = toupper((unsigned char)p[i]);
    printf("메모리 안 내용: %.*s", (int)st.st_size, p);

    if (!priv)
        msync(p, st.st_size, MS_SYNC);          /* 디스크에 즉시 반영 */
    munmap(p, st.st_size);
    printf("모드: %s\n", priv ? "MAP_PRIVATE (파일은 그대로)" : "MAP_SHARED (파일에 반영)");
    return 0;
}

🎬 D3-5 Copy-on-Write — 자식이 쓰는 비율을 바꾸며 복사되는 페이지 보기

5-1. fork 는 메모리를 복사하지 않는다

Day 1 에서 fork 후 자식의 x 와 부모의 x 가 달랐다. 그렇다고 fork 가 50MB 를 그 자리에서 복사하는 것은 아니다.

  1. fork 직후 자식은 부모와 같은 물리 페이지를 가리키고, 두 쪽 모두 읽기 전용으로 표시된다.
  2. 누군가 쓰려고 하면 page fault → 커널이 그 페이지만 복사해 쓰는 쪽에 준다.
  3. 쓰지 않은 페이지는 끝까지 공유된다.

그래서 fork 는 빠르고, fork 직후 exec 하는 셸은 거의 복사를 하지 않는다.

5-2. 예제 05 — 쓴 만큼만 복사

src/05_cow.c

/* 05_cow.c — fork 후 Copy-on-Write: 자식이 쓴 페이지만 복사된다
 * 빌드: make 05_cow
 * 실행: ./05_cow 0 / 10 / 50 / 100      (자식이 버퍼의 몇 % 를 쓸지)
 */
#include "memutil.h"
#include <unistd.h>
#include <sys/wait.h>

#define SIZE (50L * 1024 * 1024)

int main(int argc, char *argv[])
{
    int pct = (argc > 1) ? atoi(argv[1]) : 50;
    long page = sysconf(_SC_PAGESIZE);
    char *buf = malloc(SIZE);
    if (buf == NULL)
        return 1;
    memset(buf, 1, SIZE);                       /* 부모가 미리 모두 채움 */

    pid_t pid = fork();
    if (pid == 0) {
        long f0 = minflt();
        long limit = SIZE * pct / 100;
        for (long i = 0; i < limit; i += page)
            buf[i] = 2;                         /* 쓰는 순간 그 페이지만 복사 */
        long faults = minflt() - f0;
        printf("자식이 %3d %% (%5ld 페이지) 씀 → minflt %+ld\n", pct, limit / page, faults);
        exit(0);
    }
    wait(NULL);
    printf("부모 buf[0] = %d (자식이 써도 부모 값은 그대로)\n", buf[0]);
    free(buf);
    return 0;
}

자식이   0 % (    0 페이지) 씀 → minflt +0
자식이  10 % ( 1280 페이지) 씀 → minflt +1280
자식이  50 % ( 6400 페이지) 씀 → minflt +6400
자식이 100 % (12800 페이지) 씀 → minflt +12800
부모 buf[0] = 1 (자식이 써도 부모 값은 그대로)

fault 수가 쓴 페이지 수와 정확히 같다.

6-1. mlockall — 실시간 태스크의 page fault 없애기

Day 2 의 jitter 는 "깨어나는 지연" 을 쟀다. 깨어난 뒤 처음 만지는 메모리 에서 page fault 가 나면 그것도 지연이다. mlockall(MCL_CURRENT | MCL_FUTURE) 는 지금과 앞으로의 모든 페이지를 RAM 에 확보하고 내보내지(swap) 않게 한다.

src/06_mlock.c

/* 06_mlock.c — mlockall 로 page fault 를 미리 없애기
 * 빌드: make 06_mlock
 * 실행: ./06_mlock            (잠그지 않음)
 *       sudo ./06_mlock lock  (mlockall(MCL_CURRENT | MCL_FUTURE))
 *       ulimit -l             (일반 사용자가 잠글 수 있는 한도, kB)
 * 실시간 루프 안에서 처음 만지는 메모리가 page fault 를 일으키는지 본다.
 */
#include "memutil.h"
#include <sys/mman.h>

#define SIZE (32L * 1024 * 1024)

int main(int argc, char *argv[])
{
    int lock = (argc > 1 && strcmp(argv[1], "lock") == 0);
    if (lock && mlockall(MCL_CURRENT | MCL_FUTURE) == -1) {
        perror("mlockall (sudo 필요)");
        return 1;
    }
    long f0 = minflt();
    char *buf = malloc(SIZE);                  /* MCL_FUTURE: 할당 즉시 실제 페이지 확보 */
    if (buf == NULL)
        return 1;
    long f1 = minflt();

    double t0 = now_ms();
    for (long i = 0; i < SIZE; i += 4096)      /* "실시간 루프" 에서 처음 만짐 */
        buf[i] = 1;
    double t1 = now_ms();
    long f2 = minflt();

    printf("%-12s malloc 중 minflt %+6ld | 루프 중 minflt %+6ld | 루프 시간 %6.2f ms | VmLck %ld kB\n",
           lock ? "mlockall" : "잠금 없음", f1 - f0, f2 - f1, t1 - t0, status_kb("VmLck"));
    free(buf);
    return 0;
}

6-2. 오버커밋과 OOM killer

리눅스는 기본적으로 실제 RAM 보다 많은 malloc 을 허락한다(overcommit). 대부분의 프로그램이 받은 메모리를 다 쓰지 않기 때문이다. 정말로 RAM 이 바닥나면 OOM(Out Of Memory) killer 가 점수(/proc/PID/oom_score)가 높은 프로세스를 골라 SIGKILL 로 끝낸다.

/proc/sys/vm/overcommit_memory 정책
0 (기본) 터무니없는 요청만 거절, 나머지 허용
1 항상 허용
2 엄격 — 한도(swap + RAM × ratio)를 넘으면 malloc 이 NULL

VM 전체를 OOM 으로 몰면 sshd 까지 죽을 수 있다. 그래서 cgroup 한도 안에서만 실험한다 (Day 2 6교시와 같은 방법).

src/07_oom.c

/* 07_oom.c — 메모리를 계속 쓰다가 한도를 넘으면 OOM killer 에게 죽는다
 * 빌드: make 07_oom
 * 실행 (반드시 cgroup 한도 안에서!):
 *   sudo systemd-run --scope -p MemoryMax=100M -p MemorySwapMax=0 ./07_oom
 *   sudo dmesg | tail -5        (oom-kill 기록 확인)
 */
#include "memutil.h"
#include <unistd.h>

#define CHUNK (10L * 1024 * 1024)

int main(void)
{
    FILE *fp = fopen("/proc/sys/vm/overcommit_memory", "r");
    int oc = -1;
    if (fp) {
        if (fscanf(fp, "%d", &oc) != 1)
            oc = -1;
        fclose(fp);
    }
    printf("overcommit_memory = %d (0=추정 허용, 1=항상 허용, 2=엄격)\n", oc);

    for (int i = 1; i <= 100; i++) {
        char *p = malloc(CHUNK);
        if (p == NULL) {
            printf("malloc 실패 (%d MB 째)\n", i * 10);
            return 1;
        }
        memset(p, 1, CHUNK);                    /* 실제로 써야 RSS 가 는다 */
        printf("%4d MB 사용, VmRSS %6ld kB\n", i * 10, status_kb("VmRSS"));
        fflush(stdout);
        usleep(100000);
    }
    return 0;
}

7-1. 네 가지 대표 오류

오류 예 결과
누수 (leak) malloc 후 free 를 잊음 오래 도는 프로그램의 메모리가 계속 는다

버퍼 오버플로 8바이트에 10바이트 복사 이웃 블록 · malloc 관리 정보 파괴

7-2. 예제 08 과 두 도구

src/08_bugs.c

/* 08_bugs.c — 메모리 오류 4가지를 일부러 만들고 도구로 잡기
 * 빌드: make 08_bugs        (일반)
 *       make asan           (AddressSanitizer → 08_bugs_asan)
 * 실행: valgrind --leak-check=full ./08_bugs leak
 *       ./08_bugs_asan uaf | overflow | double
 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

static void leak(void)
{
    char *p = malloc(100);
    strcpy(p, "free 를 잊었다");
    printf("%s\n", p);
}                                       /* p 를 잃어버림 → 100 바이트 누수 */

static void use_after_free(void)
{
    int *p = malloc(4 * sizeof(int));
    p[0] = 10;
    free(p);
    printf("해제 후 읽기: %d\n", p[0]);   /* 이미 돌려준 메모리 */
}

static void overflow(void)
{
    char *p = malloc(8);
    strcpy(p, "123456789");             /* 10 바이트(널 포함)를 8 바이트에 */
    printf("%s\n", p);
    free(p);
}

static void double_free(void)
{
    char *p = malloc(16);
    free(p);
    free(p);                            /* 두 번 해제 */
}

int main(int argc, char *argv[])
{
    const char *k = (argc > 1) ? argv[1] : "leak";
    if (strcmp(k, "leak") == 0) leak();
    else if (strcmp(k, "uaf") == 0) use_after_free();
    else if (strcmp(k, "overflow") == 0) overflow();
    else if (strcmp(k, "double") == 0) double_free();
    else fprintf(stderr, "사용법: %s leak|uaf|overflow|double\n", argv[0]);
    return 0;
}

오늘 실험 네 가지를 memlab 하나로 묶었다. 실험마다 page fault · RSS · 시간을 CSV 한 줄로 출력하므로 여러 번 반복해 비교하기 쉽다.

src/memlab.c

/* memlab.c — Day 3 종합: 실험을 골라 page fault · RSS · 시간을 CSV 한 줄로
 * 빌드: make memlab
 * 실행: ./memlab demand | cow | mmap | lock      (lock 은 sudo)
 *       ./memlab -h
 * 출력: exp,size_mb,minflt,majflt,rss_kb,ms
 */
#include "memutil.h"
#include <fcntl.h>
#include <unistd.h>
#include <sys/mman.h>
#include <sys/wait.h>

#define MB (1024L * 1024)
static long PAGE;

static void touch(char *p, long size)
{
    for (long i = 0; i < size; i += PAGE)
        p[i] = 1;
}

static void row(const char *exp, long mb, long f0, long j0, double t0)
{
    printf("%s,%ld,%ld,%ld,%ld,%.2f\n", exp, mb, minflt() - f0, majflt() - j0,
           status_kb("VmRSS"), now_ms() - t0);
}

int main(int argc, char *argv[])
{
    const char *e = (argc > 1) ? argv[1] : "-h";
    long mb = (argc > 2) ? atol(argv[2]) : 64;
    long size = mb * MB;
    PAGE = sysconf(_SC_PAGESIZE);

    if (strcmp(e, "demand") == 0) {             /* malloc 후 처음 만질 때 fault */
        long f0 = minflt(), j0 = majflt(); double t0 = now_ms();
        char *p = malloc(size);
        touch(p, size);
        row(e, mb, f0, j0, t0);
        free(p);
    } else if (strcmp(e, "cow") == 0) {         /* fork 후 자식이 전부 쓸 때 fault */
        char *p = malloc(size);
        touch(p, size);
        if (fork() == 0) {
            long f0 = minflt(), j0 = majflt(); double t0 = now_ms();
            touch(p, size);
            row(e, mb, f0, j0, t0);
            exit(0);
        }
        wait(NULL);
        free(p);
    } else if (strcmp(e, "mmap") == 0) {        /* 파일 매핑을 처음 읽을 때 fault */
        char path[] = "/tmp/memlab_XXXXXX";
        int fd = mkstemp(path);
        if (fd == -1 || ftruncate(fd, size) == -1) { perror("file"); return 1; }
        char *p = mmap(NULL, size, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
        long f0 = minflt(), j0 = majflt(); double t0 = now_ms();
        touch(p, size);
        row(e, mb, f0, j0, t0);
        munmap(p, size);
        close(fd);
        unlink(path);
    } else if (strcmp(e, "lock") == 0) {        /* mlockall 후에는 루프 중 fault 없음 */
        if (mlockall(MCL_CURRENT | MCL_FUTURE) == -1) { perror("mlockall (sudo)"); return 1; }
        char *p = malloc(size);
        long f0 = minflt(), j0 = majflt(); double t0 = now_ms();
        touch(p, size);
        row(e, mb, f0, j0, t0);
        free(p);
    } else {
        fprintf(stderr, "사용법: %s demand|cow|mmap|lock [MB(기본 64)]\n", argv[0]);
        fprintf(stderr, "출력: exp,size_mb,minflt,majflt,rss_kb,ms\n");
        return 1;
    }
    return 0;
}
