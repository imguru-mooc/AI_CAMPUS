1-2. 프로세스와 쓰레드

프로세스 (fork) 쓰레드 (pthread_create)

리눅스 커널에게 쓰레드는 주소 공간을 공유하는 태스크다. 스케줄러는 쓰레드를 하나씩 따로 스케줄한다 — Day 2 의 nice · affinity · SCHED_FIFO 를 쓰레드마다 줄 수 있다.

src/01_threads.c

/* 01_threads.c — 쓰레드는 주소 공간을 공유한다 (fork 와 비교)
 * 빌드: make 01_threads
 * 실행: ./01_threads
 *       (다른 창) ps -L -p $(pgrep 01_threads)   또는  ls /proc/$(pgrep 01_threads)/task
 */
#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <unistd.h>
#include <sys/wait.h>

int g_value = 0;                         /* 모든 쓰레드가 같은 변수를 본다 */

static void *worker(void *arg)
{
    int id = *(int *)arg;
    g_value += 10;                       /* (이 예제는 순서대로 실행되도록 아래에서 join) */
    printf("쓰레드 %d: PID=%d TID=%d  &g_value=%p  g_value=%d\n",
           id, getpid(), gettid(), (void *)&g_value, g_value);
    sleep(2);                            /* ps -L 로 볼 시간 */
    int *ret = malloc(sizeof(int));
    *ret = id * 100;
    return ret;                          /* pthread_join 으로 받는다 */
}

int main(void)
{
    pthread_t th[3];
    int ids[3] = { 1, 2, 3 };
    printf("main   : PID=%d TID=%d\n", getpid(), gettid());
    for (int i = 0; i < 3; i++) {
        pthread_create(&th[i], NULL, worker, &ids[i]);
        usleep(100000);                  /* 출력이 섞이지 않게 잠깐 */
    }
    for (int i = 0; i < 3; i++) {
        void *r;
        pthread_join(th[i], &r);         /* 끝날 때까지 기다리고 반환값 회수 */
        printf("join   : 쓰레드 %d 반환값 %d\n", i + 1, *(int *)r);
        free(r);
    }
    printf("쓰레드가 모두 끝난 뒤 g_value = %d (공유)\n", g_value);

    fflush(stdout);                      /* Day 1: fork 전에 stdio 버퍼를 비운다 */
    pid_t pid = fork();                  /* 비교: 프로세스는 복사본 */
    if (pid == 0) {
        g_value += 1000;
        printf("fork 자식: g_value = %d\n", g_value);
        exit(0);
    }
    wait(NULL);
    printf("fork 부모: g_value = %d (자식이 바꿔도 그대로)\n", g_value);
    return 0;
}

학습 목표

🎬 D5-2 race condition — 두 쓰레드의 읽기 · 더하기 · 쓰기를 한 단계씩 끼워 넣기

2-1. counter++ 는 한 번에 일어나지 않는다

counter++ 는 기계어로 읽기 → 더하기 → 쓰기 세 단계다. 두 쓰레드가 끼어들면 한쪽의 더하기가 사라진다.

counter = 5
  쓰레드 A: 읽기 (5)
  쓰레드 B: 읽기 (5)
  쓰레드 A: 더하기 → 6, 쓰기 (6)
  쓰레드 B: 더하기 → 6, 쓰기 (6)     ← 두 번 더했는데 6

이렇게 결과가 실행 순서(타이밍)에 따라 달라지는 상황을 race condition, 공유 데이터를 고치는 코드 구간을 임계 구역(critical section) 이라 한다.

src/02_race.c

/* 02_race.c — 공유 카운터를 여러 쓰레드가 증가시키면?
 * 빌드: make 02_race
 * 실행: ./02_race 2 1000000 none      (보호 없음 — 결과가 매번 다르다)
 *       ./02_race 2 1000000 mutex     (pthread_mutex)
 *       ./02_race 2 1000000 atomic    (원자적 덧셈)
 *       ./02_race 2 1000000 spin      (spinlock)
 *       objdump -d 02_race | grep -A6 "<add_none>"   (counter++ 의 기계어)
 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>
#include <pthread.h>

static volatile long counter = 0;     /* volatile: 매번 메모리에서 읽고 쓰게 해 race 를 드러낸다 */
static long iters;
static pthread_mutex_t mtx = PTHREAD_MUTEX_INITIALIZER;
static pthread_spinlock_t spin;

__attribute__((noinline)) static void add_none(void) { counter++; }   /* 읽기(mov) → 더하기(add) → 쓰기(mov) */

static void *w_none(void *a)   { (void)a; for (long i = 0; i < iters; i++) add_none(); return NULL; }
static void *w_mutex(void *a)  { (void)a; for (long i = 0; i < iters; i++) { pthread_mutex_lock(&mtx); counter++; pthread_mutex_unlock(&mtx); } return NULL; }
static void *w_atomic(void *a) { (void)a; for (long i = 0; i < iters; i++) __atomic_fetch_add((long *)&counter, 1, __ATOMIC_RELAXED); return NULL; }
static void *w_spin(void *a)   { (void)a; for (long i = 0; i < iters; i++) { pthread_spin_lock(&spin); counter++; pthread_spin_unlock(&spin); } return NULL; }

static double now_ms(void)
{
    struct timespec t;
    clock_gettime(CLOCK_MONOTONIC, &t);
    return t.tv_sec * 1000.0 + t.tv_nsec / 1e6;
}

int main(int argc, char *argv[])
{
    int n = (argc > 1) ? atoi(argv[1]) : 2;
    iters = (argc > 2) ? atol(argv[2]) : 1000000;
    const char *mode = (argc > 3) ? argv[3] : "none";
    void *(*fn)(void *) = w_none;
    if (strcmp(mode, "mutex") == 0) fn = w_mutex;
    else if (strcmp(mode, "atomic") == 0) fn = w_atomic;
    else if (strcmp(mode, "spin") == 0) fn = w_spin;
    pthread_spin_init(&spin, PTHREAD_PROCESS_PRIVATE);

    pthread_t th[16];
    if (n > 16) n = 16;
    double t0 = now_ms();
    for (int i = 0; i < n; i++)
        pthread_create(&th[i], NULL, fn, NULL);
    for (int i = 0; i < n; i++)
        pthread_join(th[i], NULL);
    double ms = now_ms() - t0;
    long expect = n * iters;
    printf("%-6s 쓰레드 %d × %ld : 기대 %ld, 실제 %ld (잃어버림 %ld)  %.1f ms\n",
           mode, n, iters, expect, counter, expect - counter, ms);
    return 0;
}

🎬 D5-3 mutex · deadlock — 잠금 대기열과 반대 순서 잠금의 교착

3-1. 한 번에 한 쓰레드만

mutex(상호 배제) 는 잠금이다. pthread_mutex_lock 을 먼저 잡은 쓰레드만 임계 구역에 들어가고, 나머지는 unlock 될 때까지 잠든다.

pthread_mutex_lock(&mtx);
counter++;                 /* 임계 구역 — 짧게! */
pthread_mutex_unlock(&mtx);

정확하지만 8배 느리다 (5.3 → 39 ms). 잠금 · 해제도 비용이고, 경쟁하면 잠들었다 깨는 비용(Day 2 의 컨텍스트 스위치)이 더해진다. 그래서 임계 구역은 꼭 필요한 줄만, 오래 걸리는 계산 · I/O 는 잠금 밖으로 뺀다.

3-2. 교착(deadlock)

두 쓰레드가 mutex 두 개를 반대 순서로 잡으면, 서로 상대가 가진 것을 기다리며 영원히 멈춘다.

t1: A 잡음 → B 기다림 …
t2: B 잡음 → A 기다림 …     ← 둘 다 진행 불가

src/03_deadlock.c

/* 03_deadlock.c — 두 mutex 를 반대 순서로 잡으면 교착(deadlock)
 * 빌드: make 03_deadlock
 * 실행: ./03_deadlock          (A→B / B→A — 멈춘다, 3초 뒤 감시자가 알림)
 *       ./03_deadlock fixed    (둘 다 A→B — 정상 종료)
 *       멈춘 상태에서 다른 창: sudo gdb -p $(pgrep 03_deadlock) -batch -ex "thread apply all bt"
 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <signal.h>
#include <pthread.h>
#include <unistd.h>

static pthread_mutex_t A = PTHREAD_MUTEX_INITIALIZER;
static pthread_mutex_t B = PTHREAD_MUTEX_INITIALIZER;
static int fixed = 0;

static void *t1(void *arg)
{
    (void)arg;
    pthread_mutex_lock(&A);
    printf("t1: A 잡음\n");
    usleep(100000);                      /* 상대가 B 를 잡을 시간 */
    printf("t1: B 기다림...\n");
    pthread_mutex_lock(&B);
    printf("t1: A, B 모두 잡음 — 작업\n");
    pthread_mutex_unlock(&B);
    pthread_mutex_unlock(&A);
    return NULL;
}

static void *t2(void *arg)
{
    (void)arg;
    pthread_mutex_t *first = fixed ? &A : &B, *second = fixed ? &B : &A;
    pthread_mutex_lock(first);
    printf("t2: %s 잡음\n", fixed ? "A" : "B");
    usleep(100000);
    printf("t2: %s 기다림...\n", fixed ? "B" : "A");
    pthread_mutex_lock(second);
    printf("t2: A, B 모두 잡음 — 작업\n");
    pthread_mutex_unlock(second);
    pthread_mutex_unlock(first);
    return NULL;
}

static void on_alarm(int sig)
{
    (void)sig;
    const char msg[] = "감시자: 3초 동안 진행 없음 — 교착(deadlock). 서로 상대가 가진 mutex 를 기다린다.\n";
    if (write(2, msg, sizeof(msg) - 1) < 0) { }
    _exit(1);
}

int main(int argc, char *argv[])
{
    fixed = (argc > 1 && strcmp(argv[1], "fixed") == 0);
    signal(SIGALRM, on_alarm);
    alarm(3);                            /* 3초 안에 안 끝나면 교착으로 본다 */
    pthread_t a, b;
    pthread_create(&a, NULL, t1, NULL);
    pthread_create(&b, NULL, t2, NULL);
    pthread_join(a, NULL);
    pthread_join(b, NULL);
    printf("정상 종료 (%s)\n", fixed ? "잠금 순서를 A→B 로 통일" : "운 좋게 엇갈리지 않음");
    return 0;
}

4-1. 생산자 · 소비자

센서 쓰레드가 값을 만들고(생산), 처리 쓰레드가 꺼내 쓴다(소비). 사이에 링 버퍼를 둔다. 버퍼가 비었으면 소비자가, 가득 찼으면 생산자가 기다려야 한다. 반복해서 확인(바쁜 대기)하면 CPU 를 낭비하므로 조건 변수로 잠든다.

함수 하는 일
pthread_cond_wait(&c, &m) m 을 풀고 잠든다 → 깨어나면 m 을 다시 잡고 돌아온다
pthread_cond_signal(&c) c 에서 기다리는 쓰레드 하나를 깨운다
pthread_cond_broadcast(&c) 모두 깨운다 (종료 알림 등)

반드시 while 로 조건을 다시 확인한다. 깨어난 사이 다른 쓰레드가 먼저 값을 가져갔을 수 있고, 이유 없이 깨는 경우(spurious wakeup)도 표준이 허용한다.

src/04_prodcons.c

/* 04_prodcons.c — 생산자 · 소비자와 조건 변수 (링 버퍼 8칸)
 * 빌드: make 04_prodcons
 * 실행: ./04_prodcons            (소비자가 느림 → 생산자가 "가득 참" 에서 기다림)
 *       ./04_prodcons fastcons   (생산자가 느림 → 소비자가 "비어 있음" 에서 기다림)
 */
#include <stdio.h>
#include <string.h>
#include <pthread.h>
#include <unistd.h>

#define CAP 8
#define N   40

static int buf[CAP];
static int head = 0, tail = 0, count = 0;
static pthread_mutex_t m = PTHREAD_MUTEX_INITIALIZER;
static pthread_cond_t not_full  = PTHREAD_COND_INITIALIZER;
static pthread_cond_t not_empty = PTHREAD_COND_INITIALIZER;
static int prod_wait = 0, cons_wait = 0;
static int slow_cons = 1;

static void *producer(void *arg)
{
    (void)arg;
    for (int i = 1; i <= N; i++) {
        if (!slow_cons) usleep(3000);
        pthread_mutex_lock(&m);
        while (count == CAP) {           /* if 가 아니라 while — 깨어나면 다시 확인 */
            prod_wait++;
            pthread_cond_wait(&not_full, &m);   /* m 을 풀고 잠듦 → 깨면 다시 잠금 */
        }
        buf[tail] = i;
        tail = (tail + 1) % CAP;
        count++;
        pthread_cond_signal(&not_empty); /* 기다리는 소비자 하나 깨움 */
        pthread_mutex_unlock(&m);
    }
    return NULL;
}

static void *consumer(void *arg)
{
    (void)arg;
    long sum = 0;
    for (int i = 1; i <= N; i++) {
        pthread_mutex_lock(&m);
        while (count == 0) {
            cons_wait++;
            pthread_cond_wait(&not_empty, &m);
        }
        int v = buf[head];
        head = (head + 1) % CAP;
        count--;
        pthread_cond_signal(&not_full);
        pthread_mutex_unlock(&m);
        sum += v;
        if (slow_cons) usleep(3000);     /* 처리하는 척 */
    }
    printf("소비자: 합계 %ld (기대 %d)\n", sum, N * (N + 1) / 2);
    return NULL;
}

int main(int argc, char *argv[])
{
    slow_cons = !(argc > 1 && strcmp(argv[1], "fastcons") == 0);
    pthread_t p, c;
    pthread_create(&p, NULL, producer, NULL);
    pthread_create(&c, NULL, consumer, NULL);
    pthread_join(p, NULL);
    pthread_join(c, NULL);
    printf("생산자가 '가득 참' 에서 기다린 횟수 %d, 소비자가 '비어 있음' 에서 기다린 횟수 %d\n", prod_wait, cons_wait);
    return 0;
}

5-1. 세마포어 — 남은 자리 세기

세마포어는 정수 카운터다. sem_wait 는 1 을 빼되 0 이면 기다리고, sem_post 는 1 을 더하며 기다리던 쓰레드를 깨운다. "동시에 N 개까지" 를 표현할 때 쓴다 — 예: 가속기(NPU) 2개를 쓰레드 5개가 나눠 쓰기.

src/05_sem.c

/* 05_sem.c — 세마포어로 동시 사용자 수 제한 (가속기 2개를 쓰레드 5개가 나눠 씀)
 * 빌드: make 05_sem
 * 실행: ./05_sem
 */
#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>
#include <unistd.h>

static sem_t slots;                      /* 남은 자리 수 */
static int active = 0, max_active = 0;
static pthread_mutex_t m = PTHREAD_MUTEX_INITIALIZER;

static void *job(void *arg)
{
    int id = *(int *)arg;
    sem_wait(&slots);                    /* 자리가 0 이면 여기서 기다림 */
    pthread_mutex_lock(&m);
    active++;
    if (active > max_active) max_active = active;
    printf("작업 %d 시작 (동시 실행 %d)\n", id, active);
    pthread_mutex_unlock(&m);

    usleep(200000);                      /* 가속기 사용 0.2초 */

    pthread_mutex_lock(&m);
    active--;
    printf("작업 %d 끝\n", id);
    pthread_mutex_unlock(&m);
    sem_post(&slots);                    /* 자리 반납 → 기다리던 쓰레드 하나 진행 */
    return NULL;
}

int main(void)
{
    sem_init(&slots, 0, 2);              /* 0 = 쓰레드끼리 공유, 초기값 2 */
    pthread_t th[5];
    int ids[5] = { 1, 2, 3, 4, 5 };
    for (int i = 0; i < 5; i++)
        pthread_create(&th[i], NULL, job, &ids[i]);
    for (int i = 0; i < 5; i++)
        pthread_join(th[i], NULL);
    printf("최대 동시 실행 = %d (세마포어 초기값 2)\n", max_active);
    sem_destroy(&slots);
    return 0;
}

커널은 쓰레드를 따로 스케줄하므로, 한 프로세스 안에서도 센서 쓰레드만 SCHED_FIFO · CPU 1, 나머지는 OTHER 로 둘 수 있다 (Day 2 8교시의 원칙).

함수 대상
pthread_setaffinity_np(th, size, &set) 그 쓰레드의 CPU affinity
pthread_setschedparam(th, SCHED_FIFO, &sp) 그 쓰레드의 정책 · 우선순위 (sudo)
ps -L -o tid,cls,rtprio,psr,comm -p PID 쓰레드별 정책(CLS) · 우선순위 · 실행 CPU(PSR)

6-2. false sharing — 다른 변수인데 서로 방해

CPU 캐시는 64 바이트 줄(cache line) 단위로 움직인다. 두 쓰레드가 서로 다른 변수를 고쳐도, 그 변수들이 같은 64 바이트 안에 있으면 캐시 줄이 두 CPU 사이를 계속 오가며 느려진다.

src/06_false_sharing.c

/* 06_false_sharing.c — 같은 캐시 라인(64B)을 두 CPU 가 고치면 느려진다
 * 빌드: make 06_false_sharing
 * 실행: ./06_false_sharing          (두 쓰레드를 CPU 0 · 1 에 고정)
 * 서로 다른 변수인데도 같은 64바이트 안에 있으면 캐시 라인이 CPU 사이를 오간다.
 */
#define _GNU_SOURCE
#include <stdio.h>
#include <time.h>
#include <pthread.h>
#include <sched.h>
#include <unistd.h>

#define ITERS 200000000L

struct together { volatile long a; volatile long b; };                          /* 같은 캐시 라인 */
struct apart    { volatile long a; char pad[64 - sizeof(long)]; volatile long b; }; /* 다른 캐시 라인 */

static struct together T;
static struct apart P __attribute__((aligned(64)));

struct arg { volatile long *p; int cpu; };

static void *spin_add(void *x)
{
    struct arg *a = x;
    cpu_set_t set;
    CPU_ZERO(&set);
    CPU_SET(a->cpu, &set);
    pthread_setaffinity_np(pthread_self(), sizeof(set), &set);   /* 쓰레드별 CPU 고정 */
    for (long i = 0; i < ITERS; i++)
        (*a->p)++;
    return NULL;
}

static double run(volatile long *x, volatile long *y)
{
    int ncpu = sysconf(_SC_NPROCESSORS_ONLN) > 1 ? 1 : 0;
    struct arg a1 = { x, 0 }, a2 = { y, ncpu };
    struct timespec s, e;
    pthread_t t1, t2;
    clock_gettime(CLOCK_MONOTONIC, &s);
    pthread_create(&t1, NULL, spin_add, &a1);
    pthread_create(&t2, NULL, spin_add, &a2);
    pthread_join(t1, NULL);
    pthread_join(t2, NULL);
    clock_gettime(CLOCK_MONOTONIC, &e);
    return (e.tv_sec - s.tv_sec) * 1000.0 + (e.tv_nsec - s.tv_nsec) / 1e6;
}

int main(void)
{
    printf("&T.a=%p &T.b=%p (거리 %ld B)\n", (void *)&T.a, (void *)&T.b, (long)((char *)&T.b - (char *)&T.a));
    printf("&P.a=%p &P.b=%p (거리 %ld B)\n", (void *)&P.a, (void *)&P.b, (long)((char *)&P.b - (char *)&P.a));
    double t1 = run(&T.a, &T.b);
    double t2 = run(&P.a, &P.b);
    printf("같은 캐시 라인 : %8.1f ms\n", t1);
    printf("64B 떨어뜨림   : %8.1f ms  (%.1f 배 빠름)\n", t2, t1 / t2);
    if (sysconf(_SC_NPROCESSORS_ONLN) < 2)
        printf("(CPU 가 1개라 두 쓰레드가 번갈아 실행 — 차이가 거의 없다)\n");
    return 0;
}

&P.a=0x563d34720080 &P.b=0x563d347200c0 (거리 64 B)
같은 캐시 라인 :    905.3 ms
64B 떨어뜨림   :    251.7 ms  (3.6 배 빠름)

쓰레드별 통계 구조체 · 카운터는 __attribute__((aligned(64))) 이나 패딩으로 캐시 줄을 나눈다.

src/edge_pipeline.c

/* edge_pipeline.c — 5일 종합: 센서 → 처리 → 기록 3단 쓰레드 파이프라인
 *
 *   [센서 쓰레드]  10ms 절대 주기로 가상 센서 값 생성 (Day 2: clock_nanosleep, FIFO, CPU 고정)
 *        │  링 버퍼 64칸 (Day 5: mutex + 조건 변수)
 *   [처리 쓰레드]  이동 평균 10개, 임계치 넘으면 알람 (Day 5: cond_wait)
 *        │  통계 구조체 (mutex)
 *   [기록 쓰레드]  1초마다 CSV 한 줄 + fsync (Day 4)
 *   [main]         mlockall (Day 3), SIGINT 로 정상 종료 (Day 1)
 *
 * 빌드: make edge_pipeline
 * 실행: ./edge_pipeline -d 10                 (일반 정책, 10초)
 *       sudo ./edge_pipeline -d 10 -r         (센서 SCHED_FIFO 80 + CPU 1 + mlockall)
 *       stress-ng --cpu 2 & 을 켜고 두 실행의 max_lat_us 를 비교
 *       Ctrl+C 로 언제든 정상 종료 (CSV 는 끝까지 저장)
 * 출력: edge.csv — sec,samples,avg,max_lat_us,alarms,dropped
 */
#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>
#include <signal.h>
#include <time.h>
#include <pthread.h>
#include <sched.h>
#include <unistd.h>
#include <sys/mman.h>

#define CAP       64
#define PERIOD_NS 10000000L          /* 10 ms */
#define WIN       10                 /* 이동 평균 창 */

static volatile sig_atomic_t g_stop = 0;
static void on_sigint(int sig) { (void)sig; g_stop = 1; }

/* ── 링 버퍼 ── */
static double ring[CAP];
static int head = 0, tail = 0, count = 0;
static pthread_mutex_t rmx = PTHREAD_MUTEX_INITIALIZER;
static pthread_cond_t  not_empty = PTHREAD_COND_INITIALIZER;

/* ── 1초 통계 (처리 · 센서 → 기록) ── */
static struct { long samples, alarms, dropped; double sum; long max_lat_ns; } st;
static pthread_mutex_t smx = PTHREAD_MUTEX_INITIALIZER;

static double threshold = 0.8;
static int use_rt = 0;
static const char *outpath = "edge.csv";

static long long ns(struct timespec t) { return t.tv_sec * 1000000000LL + t.tv_nsec; }
static void add_ns(struct timespec *t, long d)
{
    t->tv_nsec += d;
    while (t->tv_nsec >= 1000000000L) { t->tv_nsec -= 1000000000L; t->tv_sec++; }
}

static void *sensor(void *arg)
{
    (void)arg;
    if (use_rt) {                                    /* Day 2 */
        cpu_set_t set;
        CPU_ZERO(&set);
        CPU_SET(sysconf(_SC_NPROCESSORS_ONLN) > 1 ? 1 : 0, &set);
        pthread_setaffinity_np(pthread_self(), sizeof(set), &set);
        struct sched_param sp = { .sched_priority = 80 };
        if (pthread_setschedparam(pthread_self(), SCHED_FIFO, &sp) != 0)
            fprintf(stderr, "센서: SCHED_FIFO 실패 (sudo 필요) — 일반 정책으로 계속\n");
    }
    struct timespec next, now;
    clock_gettime(CLOCK_MONOTONIC, &next);
    long k = 0;
    while (!g_stop) {
        add_ns(&next, PERIOD_NS);
        clock_nanosleep(CLOCK_MONOTONIC, TIMER_ABSTIME, &next, NULL);
        clock_gettime(CLOCK_MONOTONIC, &now);
        long lat = (long)(ns(now) - ns(next));       /* 깨어남 지연 */

        double t = k++ * 0.01;                       /* 가상 센서: 2초 주기 사인파 + 잡음 */
        double v = sin(2 * M_PI * t / 2.0) + ((rand() % 1000) / 1000.0 - 0.5) * 0.3;

        pthread_mutex_lock(&rmx);
        if (count == CAP) {                          /* 가득 차면 가장 오래된 값을 버림 */
            head = (head + 1) % CAP;
            count--;
            pthread_mutex_lock(&smx); st.dropped++; pthread_mutex_unlock(&smx);
        }
        ring[tail] = v;
        tail = (tail + 1) % CAP;
        count++;
        pthread_cond_signal(&not_empty);
        pthread_mutex_unlock(&rmx);

        pthread_mutex_lock(&smx);
        if (lat > st.max_lat_ns) st.max_lat_ns = lat;
        pthread_mutex_unlock(&smx);
    }
    pthread_mutex_lock(&rmx);
    pthread_cond_broadcast(&not_empty);              /* 처리 쓰레드를 깨워 끝내게 */
    pthread_mutex_unlock(&rmx);
    return NULL;
}

static void *process(void *arg)
{
    (void)arg;
    double win[WIN] = { 0 };
    int wi = 0, filled = 0;
    for (;;) {
        pthread_mutex_lock(&rmx);
        while (count == 0 && !g_stop)
            pthread_cond_wait(&not_empty, &rmx);     /* 값이 올 때까지 잠듦 */
        if (count == 0 && g_stop) {
            pthread_mutex_unlock(&rmx);
            break;
        }
        double v = ring[head];
        head = (head + 1) % CAP;
        count--;
        pthread_mutex_unlock(&rmx);                  /* 계산은 잠금 밖에서 */

        win[wi] = v;
        wi = (wi + 1) % WIN;
        if (filled < WIN) filled++;
        double avg = 0;
        for (int i = 0; i < filled; i++) avg += win[i];
        avg /= filled;

        pthread_mutex_lock(&smx);
        st.samples++;
        st.sum += avg;
        if (avg > threshold) st.alarms++;
        pthread_mutex_unlock(&smx);
    }
    return NULL;
}

static void *logger(void *arg)
{
    (void)arg;
    FILE *fp = fopen(outpath, "w");
    if (fp == NULL) { perror(outpath); g_stop = 1; return NULL; }
    fprintf(fp, "sec,samples,avg,max_lat_us,alarms,dropped\n");
    printf("%4s %8s %8s %11s %7s %8s\n", "초", "samples", "avg", "max_lat_us", "alarms", "dropped");
    struct timespec next;
    clock_gettime(CLOCK_MONOTONIC, &next);
    for (int sec = 1; !g_stop; sec++) {
        next.tv_sec += 1;
        clock_nanosleep(CLOCK_MONOTONIC, TIMER_ABSTIME, &next, NULL);
        pthread_mutex_lock(&smx);
        long n = st.samples, a = st.alarms, d = st.dropped, lat = st.max_lat_ns;
        double avg = n ? st.sum / n : 0;
        memset(&st, 0, sizeof(st));                  /* 다음 1초를 위해 초기화 */
        pthread_mutex_unlock(&smx);
        fprintf(fp, "%d,%ld,%.3f,%.1f,%ld,%ld\n", sec, n, avg, lat / 1000.0, a, d);
        fflush(fp);
        fsync(fileno(fp));                           /* Day 4: 1초에 한 번만 디스크까지 */
        printf("%4d %8ld %8.3f %11.1f %7ld %8ld\n", sec, n, avg, lat / 1000.0, a, d);
        fflush(stdout);
    }
    fclose(fp);
    return NULL;
}

int main(int argc, char *argv[])
{
    int dur = 10, opt;
    while ((opt = getopt(argc, argv, "d:rt:o:")) != -1) {
        switch (opt) {
        case 'd': dur = atoi(optarg); break;
        case 'r': use_rt = 1; break;
        case 't': threshold = atof(optarg); break;
        case 'o': outpath = optarg; break;
        default:
            fprintf(stderr, "사용법: %s [-d 초] [-r] [-t 임계치] [-o csv]\n", argv[0]);
            return 1;
        }
    }
    if (use_rt && mlockall(MCL_CURRENT | MCL_FUTURE) == -1)   /* Day 3 */
        perror("mlockall (sudo 필요) — 잠금 없이 계속");
    signal(SIGINT, on_sigint);                                 /* Day 1 */
    printf("edge_pipeline: %d초, 정책=%s, 임계치=%.2f → %s\n", dur, use_rt ? "센서 FIFO 80 + CPU 고정 + mlockall" : "일반(OTHER)", threshold, outpath);

    pthread_t ts, tp, tl;
    pthread_create(&tp, NULL, process, NULL);
    pthread_create(&tl, NULL, logger, NULL);
    pthread_create(&ts, NULL, sensor, NULL);
    for (int i = 0; i < dur * 10 && !g_stop; i++)
        usleep(100000);
    g_stop = 1;                                                /* 모든 쓰레드에 종료 알림 */
    pthread_join(ts, NULL);
    pthread_join(tp, NULL);
    pthread_join(tl, NULL);
    printf("정상 종료 — %s 저장\n", outpath);
    return 0;
}

🔒 정답 보기 8-1 
static volatile sig_atomic_t g_stop = 0;
signal(SIGINT, on_sigint);
pthread_join(ts, NULL);              /* 센서 → 처리 → 기록 순으로 기다림 */

시그널 핸들러와 쓰레드가 함께 보는 플래그는 volatile sig_atomic_t 로 선언합니다. signal(SIGINT, on_sigint) 로 Ctrl+C 를 받고, 끝낼 때는 플래그를 세운 뒤 pthread_join 으로 쓰레드를 하나씩 기다립니다.

  2. 쓰레드가 끝날 때까지 기다리는 함수는 이다.
