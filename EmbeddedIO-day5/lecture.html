🎬 D5-1 µs 를 다투는 비트 — 폴링 간격 · 부하를 바꾸며 사용자 공간 읽기가 비트를 놓치는 모습 확인

1-1. DHT11 — 선 하나에 40비트, µs 단위로

단계 누가 길이
시작 신호 Pi 가 Low 18 ms 이상
놓기 Pi 가 입력으로 (풀업이 High) 20 ~ 40 µs
응답 DHT11 Low → High 80 µs · 80 µs
비트 × 40 Low 50 µs 뒤 High 26 ~ 28 µs = 0, 70 µs = 1
데이터 습도 정수 · 소수 · 온도 정수 · 소수 · 체크섬 5바이트

0 과 1 의 차이는 약 40 µs. 한 번이라도 40 µs 이상 늦게 보면 비트가 틀리고, 체크섬에서 걸린다. 한 번 읽기 전체는 약 4 ms 로 짧지만, 그 4 ms 동안 한 번도 늦으면 안 된다.

1-2. 배선

MCU 과정에서 DHT11 은 ESP32 의 5V 레일이었지만, 오늘은 데이터선이 Pi GPIO 로 바로 오므로 + 를 3V3 에 꽂는다 (DHT11 은 3.3 ~ 5.5 V 동작).

1-3. 사용자 공간에서 직접 읽어 보기

Day 2 의 libgpiod 로 시작 신호를 보내고, 폴링(get_value + now_ns) 또는 엣지 이벤트로 High 폭을 잰다. 실패를 보려고 만든 예제다. 가끔 성공하는 것이 더 위험하다 — 시험 때는 되고 현장에서 틀린다.

src/01_dht_poll.c

/* 01_dht_poll.c — DHT11 을 사용자 공간에서 직접 읽어 보기 (실패를 보려고 만든 예제)
 *
 *   DHT11 의 비트: High 26~28 µs = 0, High 70 µs = 1  →  수십 µs 차이를 구분해야 한다
 *   ./01_dht_poll          get_value 폴링 + now_ns() 로 High 폭 재기
 *   ./01_dht_poll -e       엣지 이벤트 + 커널 타임스탬프로 재기 (비교용)
 *   ./01_dht_poll -n 20    시도 횟수 (기본 10, 2초 간격)
 *
 * 배선: DHT11 모듈 S → GPIO4 (물리 7) · + → 3V3 · − → GND
 * ⚠ dtoverlay=dht11 이 켜져 있으면 GPIO4 를 커널이 잡고 있어 busy — 이 예제는 overlay 전에 한다
 */
#include <stdlib.h>
#include <poll.h>
#include "gpio_common.h"

#define DHT_PIN 4
#define MAXE 200

static unsigned int g_pin = DHT_PIN;

/* High 펄스 폭(µs) 목록에서 마지막 40개를 비트로 */
static int decode(const long *hw, int n, int *t, int *h)
{
    if (n < 40) return -(100 + n);                  /* 비트 부족 */
    uint8_t b[5] = { 0 };
    for (int i = 0; i < 40; i++) {
        int bit = hw[n - 40 + i] > 50;              /* 50 µs 보다 길면 1 */
        b[i / 8] = (uint8_t)(b[i / 8] << 1 | bit);
    }
    if ((uint8_t)(b[0] + b[1] + b[2] + b[3]) != b[4]) return -1;   /* 체크섬 */
    *h = b[0]; *t = b[2];
    return 0;
}

static struct gpiod_line_config *cfg(int out, int edge)
{
    struct gpiod_line_settings *s = gpiod_line_settings_new();
    gpiod_line_settings_set_direction(s, out ? GPIOD_LINE_DIRECTION_OUTPUT : GPIOD_LINE_DIRECTION_INPUT);
    if (out) gpiod_line_settings_set_output_value(s, GPIOD_LINE_VALUE_INACTIVE);
    else {
        gpiod_line_settings_set_bias(s, GPIOD_LINE_BIAS_PULL_UP);
        if (edge) gpiod_line_settings_set_edge_detection(s, GPIOD_LINE_EDGE_BOTH);
    }
    struct gpiod_line_config *c = gpiod_line_config_new();
    gpiod_line_config_add_line_settings(c, &g_pin, 1, s);
    gpiod_line_settings_free(s);
    return c;
}

/* 한 번 읽기: 0 성공 · -1 체크섬 · -1xx 비트 부족 */
static int read_once(struct gpiod_line_request *req, int use_edge, struct gpiod_edge_event_buffer *buf, int *t, int *h)
{
    struct gpiod_line_config *out = cfg(1, 0), *in = cfg(0, use_edge);
    gpiod_line_request_reconfigure_lines(req, out);         /* 시작 신호: Low 20 ms */
    sleep_ms(20);
    gpiod_line_request_reconfigure_lines(req, in);          /* 놓기 → 센서가 응답 */
    long hw[MAXE]; int n = 0;

    if (!use_edge) {                                        /* ① 폴링 */
        long long t0 = now_ns(), rise = 0;
        int last = 1;
        while (now_ns() - t0 < 8000000LL && n < MAXE) {     /* 8 ms 동안 계속 읽기 */
            int v = gpiod_line_request_get_value(req, g_pin) == GPIOD_LINE_VALUE_ACTIVE;
            if (v != last) {
                long long now = now_ns();                   /* ← 읽은 "순간" 이 아니라 지금 */
                if (v) rise = now;
                else if (rise) hw[n++] = (long)((now - rise) / 1000);
                last = v;
            }
        }
    } else {                                                /* ② 엣지 이벤트 */
        struct pollfd p = { .fd = gpiod_line_request_get_fd(req), .events = POLLIN };
        long long rise = 0;
        while (n < MAXE && poll(&p, 1, 8) > 0) {
            int m = gpiod_line_request_read_edge_events(req, buf, 64);
            for (int i = 0; i < m && n < MAXE; i++) {
                struct gpiod_edge_event *ev = gpiod_edge_event_buffer_get_event(buf, i);
                long long ts = (long long)gpiod_edge_event_get_timestamp_ns(ev);
                if (gpiod_edge_event_get_event_type(ev) == GPIOD_EDGE_EVENT_RISING_EDGE) rise = ts;
                else if (rise) hw[n++] = (long)((ts - rise) / 1000);
            }
        }
    }
    gpiod_line_config_free(out);
    gpiod_line_config_free(in);
    return decode(hw, n, t, h);
}

int main(int argc, char **argv)
{
    int use_edge = 0, tries = 10;
    for (int i = 1; i < argc; i++) {
        if (!strcmp(argv[i], "-e")) use_edge = 1;
        else if (!strcmp(argv[i], "-n") && i + 1 < argc) tries = atoi(argv[++i]);
    }
    gpio_install_sigint();
    /* 요청을 직접 만든다: 엣지 80개가 한꺼번에 오므로 커널 이벤트 버퍼를 넉넉히 (기본 = 라인당 16개) */
    struct gpiod_chip *chip = gpiod_chip_open(GPIO_CHIP);
    struct gpiod_line_config *lc = cfg(0, 0);
    struct gpiod_request_config *rc = gpiod_request_config_new();
    gpiod_request_config_set_consumer(rc, "dht-poll");
    gpiod_request_config_set_event_buffer_size(rc, 256);
    struct gpiod_line_request *req = chip ? gpiod_chip_request_lines(chip, rc, lc) : NULL;
    gpiod_request_config_free(rc);
    gpiod_line_config_free(lc);
    if (chip) gpiod_chip_close(chip);
    if (!req) perror("GPIO4 요청 (dht11 overlay 가 켜져 있으면 busy)");
    struct gpiod_edge_event_buffer *buf = gpiod_edge_event_buffer_new(64);
    if (!req || !buf) return EXIT_FAILURE;

    printf("DHT11 사용자 공간 읽기 (%s) × %d — Ctrl+C 로 종료\n", use_edge ? "엣지 타임스탬프" : "폴링", tries);
    int ok = 0, done = 0;
    for (; done < tries && !g_stop; done++) {
        int t, h, r = read_once(req, use_edge, buf, &t, &h);
        if (r == 0) { ok++; printf("%2d  OK      %d °C  %d %%\n", done + 1, t, h); }
        else if (r == -1) printf("%2d  실패    체크섬 오류 (비트 하나 이상 잘못 읽음)\n", done + 1);
        else printf("%2d  실패    비트 부족 (%d / 40)\n", done + 1, -r - 100);
        sleep_ms(2000);                                     /* DHT11 은 1초에 한 번 이하 */
    }
    printf("성공 %d / %d\n", ok, done);
    gpiod_edge_event_buffer_free(buf);
    gpiod_line_request_release(req);
    return EXIT_SUCCESS;
}

src/dht.h

/* dht.h — 커널 dht11 드라이버(IIO) 읽기
 *
 *   config.txt: dtoverlay=dht11,gpiopin=4  → /sys/bus/iio/devices/iio:deviceN/
 *     in_temp_input               밀리 °C   (23000 = 23.0 °C)
 *     in_humidityrelative_input   밀리 %    (45000 = 45 %)
 *
 *   dht_t d;  dht_open(&d);  dht_read(&d, &t, &h);
 */
#ifndef DHT_H
#define DHT_H

typedef struct {
    char dir[96];          /* 찾은 IIO 장치 디렉터리 */
    int fails;             /* 연속 실패 횟수 */
} dht_t;

int dht_open(dht_t *d);                              /* 0 = 찾음, -1 = 없음 (overlay 확인) */
int dht_read(dht_t *d, double *temp_c, double *humi); /* 0 = 성공, -1 = 3번 재시도 후 실패 */

#endif

src/dht.c

/* dht.c — dht.h 구현 */
#include <glob.h>
#include <stdio.h>
#include <string.h>
#include <time.h>
#include <unistd.h>
#include "dht.h"

static int read_long(const char *dir, const char *name, long *v)
{
    char path[160];
    snprintf(path, sizeof path, "%s/%s", dir, name);
    FILE *f = fopen(path, "r");
    if (!f) return -1;
    int ok = fscanf(f, "%ld", v) == 1;      /* 드라이버가 실패하면 read 가 EIO · ETIMEDOUT */
    fclose(f);
    return ok ? 0 : -1;
}

int dht_open(dht_t *d)
{
    glob_t g;
    memset(d, 0, sizeof *d);
    /* 장치 번호(N)는 부팅마다 바뀔 수 있으므로 습도 파일이 있는 장치를 찾는다 */
    if (glob("/sys/bus/iio/devices/iio:device*/in_humidityrelative_input", 0, NULL, &g) != 0) return -1;
    snprintf(d->dir, sizeof d->dir, "%.*s", (int)(strrchr(g.gl_pathv[0], '/') - g.gl_pathv[0]), g.gl_pathv[0]);
    globfree(&g);
    return 0;
}

int dht_read(dht_t *d, double *temp_c, double *humi)
{
    for (int i = 0; i < 3; i++) {
        long t, h;
        if (read_long(d->dir, "in_temp_input", &t) == 0 &&
            read_long(d->dir, "in_humidityrelative_input", &h) == 0) {
            *temp_c = t / 1000.0;
            *humi = h / 1000.0;
            d->fails = 0;
            return 0;
        }
        struct timespec ts = { 0, 300000000L };    /* 0.3 초 뒤 다시 */
        nanosleep(&ts, NULL);
    }
    d->fails++;
    return -1;
}

/dts-v1/;
/plugin/;

/ {
    compatible = "brcm,bcm2711";

    fragment@0 {
        target = <&gpio>;
        __overlay__ {
            eio_key_pins: eio_key_pins {
                brcm,pins = <5>;
                brcm,function = <0>;        /* 0 = 입력 */
                brcm,pull = <2>;            /* 2 = pull-up */
            };
        };
    };

        target-path = "/";
        __overlay__ {
            eio-leds {

/* 03_led_key.c — overlay 가 만든 커널 장치 쓰기: LED 클래스 + input 이벤트
 *
 *   준비: eio.dtbo 를 overlays 에 복사, config.txt 에 dtoverlay=eio, 재부팅
 *   ./03_led_key        버튼 A 를 누를 때마다 LED 트리거를 바꾼다
 *                       none(꺼짐) → default-on → timer → heartbeat → none …
 *
 *   GPIO 를 직접 요청하지 않는다 — LED 와 버튼은 이제 커널 드라이버의 것
 */
#include <dirent.h>
#include <fcntl.h>
#include <poll.h>
#include <unistd.h>
#include <stdlib.h>
#include <linux/input.h>
#include <sys/ioctl.h>
#include "gpio_common.h"

#define LED_DIR "/sys/class/leds/eio:green"

static int write_str(const char *file, const char *s)
{
    char path[128];
    snprintf(path, sizeof path, "%s/%s", LED_DIR, file);
    int fd = open(path, O_WRONLY);
    if (fd < 0) { perror(path); return -1; }
    int r = write(fd, s, strlen(s)) < 0 ? -1 : 0;
    close(fd);
    return r;
}

/* /dev/input/event* 중 이름이 "eio-keys" 인 장치 찾기 */
static int open_keys(void)
{
    DIR *d = opendir("/dev/input");
    struct dirent *e;
    while (d && (e = readdir(d))) {
        if (strncmp(e->d_name, "event", 5)) continue;
        char path[300], name[64] = "";
        snprintf(path, sizeof path, "/dev/input/%s", e->d_name);
        int fd = open(path, O_RDONLY | O_CLOEXEC);
        if (fd < 0) continue;
        ioctl(fd, EVIOCGNAME(sizeof name), name);        /* 장치 이름 묻기 */
        if (!strcmp(name, "eio-keys")) { printf("버튼 장치: %s (%s)\n", path, name); closedir(d); return fd; }
        close(fd);
    }
    if (d) closedir(d);
    return -1;
}

int main(void)
{
    const char *TRIG[] = { "none", "default-on", "timer", "heartbeat" };
    gpio_install_sigint();
    int kfd = open_keys();
    if (kfd < 0) { fprintf(stderr, "eio-keys 없음 — overlay 적용 · 재부팅 · input 그룹 확인\n"); return EXIT_FAILURE; }
    if (write_str("trigger", "none") < 0) { fprintf(stderr, "LED 쓰기 권한 확인 (sudo 또는 udev 규칙)\n"); return EXIT_FAILURE; }

    printf("버튼 A 를 누르세요 — Ctrl+C 로 종료\n");
    int k = 0;
    struct pollfd p = { .fd = kfd, .events = POLLIN };
    while (!g_stop) {
        if (poll(&p, 1, -1) <= 0) continue;
        struct input_event ev;
        if (read(kfd, &ev, sizeof ev) != sizeof ev) continue;
        if (ev.type != EV_KEY) continue;                  /* EV_SYN 등은 건너뜀 */
        printf("%ld.%06ld  code %d  value %d (%s)\n", (long)ev.input_event_sec, (long)ev.input_event_usec,
               ev.code, ev.value, ev.value == 1 ? "누름" : ev.value == 0 ? "뗌" : "반복");
        if (ev.code == KEY_ENTER && ev.value == 1) {
            k = (k + 1) % 4;
            write_str("trigger", TRIG[k]);
            printf("  → trigger = %s\n", TRIG[k]);
        }
    }
    write_str("trigger", "heartbeat");                     /* overlay 기본값으로 되돌림 */
    close(kfd);
    printf("\n종료\n");
    return EXIT_SUCCESS;
}

  ③ 50 µs 쉼 (명령 실행 ≥ 37 µs)
명령 값 뜻
지우기 0x01 화면 비우고 커서 처음 — 1.52 ms 이상 기다림

/* lcd.c — lcd.h 구현 (HD44780 4비트)
 *
 * HD44780 의 타이밍은 "최소 시간" 만 있다 (E 폭 ≥ 450 ns, 명령 뒤 ≥ 37 µs, clear 뒤 ≥ 1.52 ms).
 * 늦어지는 것은 괜찮다 → 스케줄링 지연이 있는 사용자 공간에서도 동작한다. (DHT11 과 다른 점)
 */
#include <stdarg.h>
#include <stdio.h>
#include <stdlib.h>
#include "gpio_common.h"
#include "lcd.h"

enum { RS, E, D4, D5, D6, D7, NPIN };
static const unsigned int PINS[NPIN] = { 12, 16, 20, 21, 26, 22 };

struct lcd {
    struct gpiod_line_request *req;
    enum gpiod_line_value v[NPIN];
};

static void us_sleep(long us)
{
    struct timespec ts = { us / 1000000, (us % 1000000) * 1000 };
    while (nanosleep(&ts, &ts) < 0 && errno == EINTR) ;      /* Ctrl+C 로 짧아지지 않게 */
}

static void apply(lcd_t *l) { gpiod_line_request_set_values(l->req, l->v); }

/* 4비트 하나: D4~D7 에 올리고 E 를 High → Low (떨어지는 순간 LCD 가 읽음) */
static void nibble(lcd_t *l, uint8_t n)
{
    for (int i = 0; i < 4; i++) l->v[D4 + i] = (n >> i) & 1 ? GPIOD_LINE_VALUE_ACTIVE : GPIOD_LINE_VALUE_INACTIVE;
    l->v[E] = GPIOD_LINE_VALUE_ACTIVE;   apply(l);   us_sleep(1);
    l->v[E] = GPIOD_LINE_VALUE_INACTIVE; apply(l);   us_sleep(1);
}

static void send(lcd_t *l, uint8_t b, int rs)
{
    l->v[RS] = rs ? GPIOD_LINE_VALUE_ACTIVE : GPIOD_LINE_VALUE_INACTIVE;
    nibble(l, b >> 4);                  /* 위 4비트 먼저 */
    nibble(l, b & 0x0F);                /* 아래 4비트 */
    us_sleep(50);                       /* 명령 실행 ≥ 37 µs */
}

void lcd_cmd(lcd_t *l, uint8_t c) { send(l, c, 0); }
void lcd_putc(lcd_t *l, char c)   { send(l, (uint8_t)c, 1); }
void lcd_clear(lcd_t *l)          { lcd_cmd(l, 0x01); us_sleep(2000); }
void lcd_goto(lcd_t *l, int col, int row) { lcd_cmd(l, (uint8_t)(0x80 | ((row ? 0x40 : 0x00) + col))); }
void lcd_puts(lcd_t *l, const char *s) { while (*s) lcd_putc(l, *s++); }

void lcd_printf(lcd_t *l, int row, const char *fmt, ...)
{
    char buf[17];
    va_list ap;
    va_start(ap, fmt);
    vsnprintf(buf, sizeof buf, fmt, ap);
    va_end(ap);
    int n = (int)strlen(buf);
    for (; n < 16; n++) buf[n] = ' ';   /* 남은 칸은 공백 — 이전 글자가 남지 않게 */
    buf[16] = '\0';
    lcd_goto(l, 0, row);
    lcd_puts(l, buf);
}

void lcd_custom(lcd_t *l, int slot, const uint8_t bitmap[8])
{
    lcd_cmd(l, (uint8_t)(0x40 | ((slot & 7) << 3)));        /* CGRAM 주소 */
    for (int i = 0; i < 8; i++) send(l, bitmap[i] & 0x1F, 1);
    lcd_goto(l, 0, 0);                                       /* 다시 DDRAM 으로 */
}

lcd_t *lcd_open(void)
{
    lcd_t *l = calloc(1, sizeof *l);
    if (!l) return NULL;
    l->req = gpio_request_output("lcd", PINS, NPIN);
    if (!l->req) { free(l); return NULL; }

    us_sleep(50000);                    /* 전원 후 40 ms 이상 */
    l->v[RS] = GPIOD_LINE_VALUE_INACTIVE;
    nibble(l, 0x3); us_sleep(4500);     /* 8비트 모드라고 세 번 말해 상태를 맞춘다 */
    nibble(l, 0x3); us_sleep(150);
    nibble(l, 0x3); us_sleep(150);
    nibble(l, 0x2); us_sleep(150);      /* 이제 4비트 모드 */
    lcd_cmd(l, 0x28);                   /* 4비트 · 2줄 · 5×8 */
    lcd_cmd(l, 0x08);                   /* 화면 끔 */
    lcd_clear(l);
    lcd_cmd(l, 0x06);                   /* 쓰면 커서 오른쪽으로 */
    lcd_cmd(l, 0x0C);                   /* 화면 켬 · 커서 숨김 */
    return l;
}

void lcd_close(lcd_t *l)
{
    if (!l) return;
    lcd_clear(l);
    for (int i = 0; i < NPIN; i++) l->v[i] = GPIOD_LINE_VALUE_INACTIVE;
    apply(l);
    gpiod_line_request_release(l->req);
    free(l);
}

void lcd_release(lcd_t *l)
{
    if (!l) return;
    gpiod_line_request_release(l->req);     /* 출력은 마지막 값 그대로 — LCD 는 글자를 유지한다 */
    free(l);
}

struct lcd 의 내용은 lcd.c 안에만 있고, 헤더에는 typedef struct lcd lcd_t; 만 있다(불투명 포인터). 쓰는 쪽은 핀 번호를 몰라도 된다 — 종합 프로젝트의 모듈은 모두 이 모양이다.

/* 04_lcd_demo.c — lcd.c 모듈 시험: 글자 · 고정 폭 · 사용자 문자
 *
 *   make 04_lcd_demo   (lcd.c 와 함께 링크)
 *   ./04_lcd_demo      윗줄 "Hello, Pi 4!" · 아랫줄 카운터와 ° · 하트
 */
#include <stdlib.h>
#include "gpio_common.h"
#include "lcd.h"

static const uint8_t DEG[8]   = { 0x06, 0x09, 0x09, 0x06, 0x00, 0x00, 0x00, 0x00 };   /* ° */
static const uint8_t HEART[8] = { 0x00, 0x0A, 0x1F, 0x1F, 0x0E, 0x04, 0x00, 0x00 };

int main(void)
{
    gpio_install_sigint();
    lcd_t *l = lcd_open();
    if (!l) return EXIT_FAILURE;
    lcd_custom(l, 1, DEG);                              /* 0 번은 쓰지 않는다: 문자 0 = 문자열의 끝 */
    lcd_custom(l, 2, HEART);

    lcd_printf(l, 0, "Hello, Pi 4!");
    lcd_goto(l, 15, 0); lcd_putc(l, 2);                 /* 하트 = 사용자 문자 2 */
    printf("LCD 출력 중 — 글자가 안 보이면 가변저항(V0) 을 천천히 돌린다. Ctrl+C 로 종료\n");

    for (int n = 1200; !g_stop; n = n > 0 ? n - 7 : 1200) {
        lcd_printf(l, 1, "%4d  25.0\x01" "C", n);       /* \x01 = 사용자 문자 1 (°) — %4d 로 폭 고정 */
        sleep_ms(200);
    }
    lcd_close(l);
    printf("\n종료\n");
    return EXIT_SUCCESS;
}

/* sonar.c — sonar.h 구현: Echo 의 rising · falling 커널 타임스탬프 차이 */
#include <poll.h>
#include <stdlib.h>
#include "gpio_common.h"
#include "sonar.h"

struct sonar {
    unsigned int trig, echo;
    struct gpiod_line_request *t, *e;
    struct gpiod_edge_event_buffer *buf;
};

sonar_t *sonar_open(unsigned int trig, unsigned int echo)
{
    sonar_t *s = calloc(1, sizeof *s);
    if (!s) return NULL;
    s->trig = trig; s->echo = echo;
    s->t = gpio_request_output("sonar", &s->trig, 1);
    struct gpiod_line_settings *st = gpiod_line_settings_new();
    gpiod_line_settings_set_direction(st, GPIOD_LINE_DIRECTION_INPUT);
    gpiod_line_settings_set_edge_detection(st, GPIOD_LINE_EDGE_BOTH);
    gpiod_line_settings_set_bias(st, GPIOD_LINE_BIAS_DISABLED);
    s->e = gpio_request("sonar", &s->echo, 1, st);
    gpiod_line_settings_free(st);
    s->buf = gpiod_edge_event_buffer_new(8);
    if (!s->t || !s->e || !s->buf) { sonar_close(s); return NULL; }
    return s;
}

double sonar_measure(sonar_t *s)
{
    struct pollfd p = { .fd = gpiod_line_request_get_fd(s->e), .events = POLLIN };
    while (poll(&p, 1, 0) > 0) gpiod_line_request_read_edge_events(s->e, s->buf, 8);   /* 찌꺼기 버림 */

    gpiod_line_request_set_value(s->t, s->trig, GPIOD_LINE_VALUE_ACTIVE);
    struct timespec ts = { 0, 10000 };
    nanosleep(&ts, NULL);
    gpiod_line_request_set_value(s->t, s->trig, GPIOD_LINE_VALUE_INACTIVE);

    long long rise = 0, fall = 0, deadline = now_ns() + 60000000LL;
    while (!fall) {
        int left = (int)((deadline - now_ns()) / 1000000);
        if (left <= 0 || poll(&p, 1, left) <= 0) return -1;
        int n = gpiod_line_request_read_edge_events(s->e, s->buf, 8);
        for (int i = 0; i < n; i++) {
            struct gpiod_edge_event *ev = gpiod_edge_event_buffer_get_event(s->buf, i);
            long long t = (long long)gpiod_edge_event_get_timestamp_ns(ev);
            if (gpiod_edge_event_get_event_type(ev) == GPIOD_EDGE_EVENT_RISING_EDGE) rise = t;
            else if (rise) { fall = t; break; }
        }
    }
    double us = (fall - rise) / 1000.0;
    return us > 30000 ? -1 : us * 0.0343 / 2;
}

static int cmp(const void *a, const void *b)
{
    double x = *(const double *)a, y = *(const double *)b;
    return (x > y) - (x < y);
}

double sonar_median(sonar_t *s, int n)
{
    double v[7]; int k = 0;
    if (n > 7) n = 7;
    for (int i = 0; i < n; i++) {
        double d = sonar_measure(s);
        if (d > 0) v[k++] = d;
        sleep_ms(60);
    }
    if (!k) return -1;
    qsort(v, (size_t)k, sizeof v[0], cmp);
    return v[k / 2];
}

void sonar_close(sonar_t *s)
{
    if (!s) return;
    if (s->t) { gpiod_line_request_set_value(s->t, s->trig, GPIOD_LINE_VALUE_INACTIVE); gpiod_line_request_release(s->t); }
    if (s->e) gpiod_line_request_release(s->e);
    if (s->buf) gpiod_edge_event_buffer_free(s->buf);
    free(s);
}

/* envmon.c — 스마트 환경 모니터 (Day 5 종합 프로젝트)
 *
 *   sensor    2초마다 온습도(DHT11 · IIO) + 조도(PCF8591 IN0)
 *   proximity 약 0.3초마다 거리(HC-SR04 중앙값 3회) — 30 cm 안이면 접근 경고
 *   control   버튼 A(GPIO5) 짧게 = LCD 화면 전환 · 길게(1초) = 부저 음소거
 *             온도 ≥ 28 °C → 릴레이(팬) ON, ≤ 27 °C → OFF · 접근 → 부저 · 경보 LED
 *   display   0.5초마다(또는 화면 전환 즉시) LCD 갱신, 1분마다 envmon.csv 기록
 *
 *   ./envmon               (Ctrl+C 또는 systemctl stop → 모두 끄고 반납)
 *   ./envmon -c 경로.csv
 *
 *   핀: DHT11 4 · I2C 2/3 · 버튼 5 · LCD 12/16/20/21/26/22 · 부저 13 · LED 17(상태)/27(경보)
 *       HC-SR04 23/24 · 릴레이 25
 */
#include <pthread.h>
#include <poll.h>
#include <stdlib.h>
#include <time.h>
#include "gpio_common.h"
#include "pcf8591.h"
#include "dht.h"
#include "sonar.h"
#include "lcd.h"

#define PIN_LED_OK 17
#define PIN_LED_AL 27
#define PIN_RELAY  25
#define PIN_BUZ    13
#define PIN_BTN     5
#define FAN_ON   28.0
#define FAN_OFF  27.0
#define NEAR_CM  30.0
#define NSCREEN   4

/* ── 공유 상태: 모든 쓰레드가 이 구조체 하나만 주고받는다 ── */
typedef struct {
    double t, h;   int th_ok;      /* 온습도 · 유효 */
    int light;                     /* 조도 % (-1 = 실패) */
    double cm;                     /* 거리 (-1 = 실패) */
    int screen, mute, fan, near;
    int err_dht, err_pcf, err_sonar;
} env_t;

static struct {
    pthread_mutex_t mx;
    pthread_cond_t  cv;            /* 화면을 바로 다시 그려 달라는 신호 */
    env_t d;
} S = { PTHREAD_MUTEX_INITIALIZER, PTHREAD_COND_INITIALIZER, { .light = -1, .cm = -1 } };
static const char *g_csv = "envmon.csv";
static long long g_start;

static void log_msg(const char *fmt, const char *a)
{
    time_t now = time(NULL);
    char ts[16]; strftime(ts, sizeof ts, "%H:%M:%S", localtime(&now));
    printf("[%s] ", ts); printf(fmt, a); printf("\n");          /* journalctl 에 그대로 남는다 */
}

/* ── sensor: 온습도 + 조도 ── */
static void *sensor_thread(void *arg)
{
    (void)arg;
    dht_t d; int have_dht = dht_open(&d) == 0;
    int pcf = pcf_open("/dev/i2c-1", PCF_ADDR);
    int was_dht = 1, was_pcf = 1;
    while (!g_stop) {
        double t = 0, h = 0;
        if (!have_dht) have_dht = dht_open(&d) == 0;
        int ok = have_dht && dht_read(&d, &t, &h) == 0;
        if (pcf < 0) pcf = pcf_open("/dev/i2c-1", PCF_ADDR);
        int v = pcf >= 0 ? pcf_read(pcf, 0) : -1;

        pthread_mutex_lock(&S.mx);
        if (ok) { S.d.t = t; S.d.h = h; S.d.th_ok = 1; } else { S.d.th_ok = 0; S.d.err_dht++; }
        S.d.light = v < 0 ? -1 : v * 100 / 255;
        if (v < 0) S.d.err_pcf++;
        pthread_mutex_unlock(&S.mx);

        if (ok != was_dht) log_msg("DHT11 %s", ok ? "복구" : "읽기 실패 — 선 확인");
        if ((v >= 0) != was_pcf) log_msg("PCF8591 %s", v >= 0 ? "복구" : "응답 없음 — I2C 선 확인");
        was_dht = ok; was_pcf = v >= 0;
        for (int i = 0; i < 20 && !g_stop; i++) sleep_ms(100);   /* 2초 — Ctrl+C 에 빨리 반응 */
    }
    if (pcf >= 0) close(pcf);
    return NULL;
}

/* ── proximity: 거리 ── */
static void *proximity_thread(void *arg)
{
    sonar_t *so = arg;
    int was_near = 0;
    while (!g_stop) {
        double cm = so ? sonar_median(so, 3) : -1;
        pthread_mutex_lock(&S.mx);
        S.d.cm = cm;
        if (cm < 0) S.d.err_sonar++;
        S.d.near = cm > 0 && cm < NEAR_CM;
        int near = S.d.near;
        if (near != was_near) pthread_cond_signal(&S.cv);       /* 경고 화면을 바로 */
        pthread_mutex_unlock(&S.mx);
        if (near && !was_near) log_msg("접근 감지%s", "");
        was_near = near;
        sleep_ms(100);
    }
    return NULL;
}

/* ── control: 버튼 이벤트 + 판단 + 출력 ── */
static struct gpiod_line_request *g_out;
static const unsigned int OUTS[4] = { PIN_LED_OK, PIN_LED_AL, PIN_RELAY, PIN_BUZ };

static void *control_thread(void *arg)
{
    struct gpiod_line_request *btn = arg;
    struct gpiod_edge_event_buffer *buf = gpiod_edge_event_buffer_new(8);
    struct pollfd p = { .fd = gpiod_line_request_get_fd(btn), .events = POLLIN };
    long long down_at = 0;
    while (!g_stop) {
        if (poll(&p, 1, 50) > 0) {                               /* 50 ms 마다 출력 갱신 */
            int n = gpiod_line_request_read_edge_events(btn, buf, 8);
            for (int i = 0; i < n; i++) {
                struct gpiod_edge_event *ev = gpiod_edge_event_buffer_get_event(buf, i);
                long long ts = (long long)gpiod_edge_event_get_timestamp_ns(ev);
                if (gpiod_edge_event_get_event_type(ev) == GPIOD_EDGE_EVENT_FALLING_EDGE) down_at = ts;
                else if (down_at) {
                    long long ms = (ts - down_at) / 1000000;  down_at = 0;
                    pthread_mutex_lock(&S.mx);
                    if (ms >= 1000) S.d.mute = !S.d.mute; else S.d.screen = (S.d.screen + 1) % NSCREEN;
                    pthread_cond_signal(&S.cv);
                    pthread_mutex_unlock(&S.mx);
                    log_msg(ms >= 1000 ? "버튼 길게 — 음소거 전환%s" : "버튼 — 화면 전환%s", "");
                }
            }
        }
        pthread_mutex_lock(&S.mx);
        if (S.d.th_ok && S.d.t >= FAN_ON)  S.d.fan = 1;                  /* 히스테리시스 */
        if (S.d.th_ok && S.d.t <= FAN_OFF) S.d.fan = 0;
        int fan = S.d.fan, near = S.d.near, mute = S.d.mute, err = !S.d.th_ok || S.d.light < 0 || S.d.cm < 0;
        pthread_mutex_unlock(&S.mx);

        long long ms = (now_ns() - g_start) / 1000000;
        enum gpiod_line_value v[4] = {
            (ms % (err ? 250 : 1000)) < (err ? 125 : 100),         /* 상태 LED: 정상 1초에 한 번 · 오류 빠르게 */
            near,                                                    /* 경보 LED */
            fan,                                                     /* 릴레이 = 팬 */
            near && !mute && (ms % 500) < 100,                       /* 부저 삑 */
        };
        gpiod_line_request_set_values(g_out, v);
    }
    gpiod_edge_event_buffer_free(buf);
    return NULL;
}

/* ── display: LCD + CSV ── */
static void *display_thread(void *arg)
{
    lcd_t *l = arg;
    static const uint8_t DEG[8] = { 0x06, 0x09, 0x09, 0x06, 0, 0, 0, 0 };
    lcd_custom(l, 1, DEG);
    long long last_csv = 0;
    pthread_mutex_lock(&S.mx);
    while (!g_stop) {
        env_t c = S.d;                                           /* 잠근 채 복사 → 풀고 그린다 */
        pthread_mutex_unlock(&S.mx);

        char a[32], b[32];                                       /* lcd_printf 가 16칸으로 자른다 */
        if (c.near) { snprintf(a, sizeof a, "!! APPROACH !!"); snprintf(b, sizeof b, "dist %5.1f cm", c.cm); }
        else switch (c.screen) {
        case 0:
            if (c.th_ok) snprintf(a, sizeof a, "T %4.1f\x01" "C H %2.0f%%", c.t, c.h);
            else snprintf(a, sizeof a, "T --.-  H --");
            snprintf(b, sizeof b, "Fan %-3s Mute %s", c.fan ? "ON" : "OFF", c.mute ? "Y" : "N");
            break;
        case 1:
            if (c.light < 0) snprintf(a, sizeof a, "Light  --");
            else snprintf(a, sizeof a, "Light %3d %%", c.light);
            { int n = c.light < 0 ? 0 : c.light * 16 / 100; for (int i = 0; i < 16; i++) b[i] = i < n ? (char)0xFF : ' '; b[16] = 0; }
            break;
        case 2:
            if (c.cm < 0) snprintf(a, sizeof a, "Dist  ---.- cm");
            else snprintf(a, sizeof a, "Dist %6.1f cm", c.cm);
            snprintf(b, sizeof b, "near < %.0f cm", NEAR_CM);
            break;
        default:
            snprintf(a, sizeof a, "up %lld s", (now_ns() - g_start) / 1000000000LL);
            snprintf(b, sizeof b, "E D%d P%d S%d", c.err_dht % 100, c.err_pcf % 100, c.err_sonar % 100);
        }
        lcd_printf(l, 0, "%s", a);
        lcd_printf(l, 1, "%s", b);

        long long now = now_ns();
        if (now - last_csv >= 60000000000LL) {                   /* 1분마다 */
            last_csv = now;
            FILE *f = fopen(g_csv, "a");
            if (f) {
                time_t tt = time(NULL); char ts[32];
                strftime(ts, sizeof ts, "%Y-%m-%d %H:%M:%S", localtime(&tt));
                fprintf(f, "%s,%.1f,%.0f,%d,%.1f,%d,%d\n", ts, c.th_ok ? c.t : -1, c.th_ok ? c.h : -1,
                        c.light, c.cm, c.fan, c.near);
                fclose(f);
            }
        }
        /* 0.5초 기다리되, 화면 전환 · 접근 신호가 오면 바로 깨어난다 */
        struct timespec dl; clock_gettime(CLOCK_REALTIME, &dl);
        dl.tv_nsec += 500000000L; if (dl.tv_nsec >= 1000000000L) { dl.tv_sec++; dl.tv_nsec -= 1000000000L; }
        pthread_mutex_lock(&S.mx);
        pthread_cond_timedwait(&S.cv, &S.mx, &dl);
    }
    pthread_mutex_unlock(&S.mx);
    return NULL;
}

int main(int argc, char **argv)
{
    for (int i = 1; i < argc; i++)
        if (!strcmp(argv[i], "-c") && i + 1 < argc) g_csv = argv[++i];
    setvbuf(stdout, NULL, _IOLBF, 0);          /* systemd 아래에서는 stdout 이 파이프 → 줄 단위로 내보내기 */
    gpio_install_sigint();                     /* SIGINT · SIGTERM(systemctl stop) → g_stop */
    g_start = now_ns();

    lcd_t *lcd = lcd_open();
    g_out = gpio_request_output("envmon", OUTS, 4);
    unsigned int bpin = PIN_BTN;
    struct gpiod_line_request *btn = gpio_request_input("envmon", &bpin, 1, GPIOD_LINE_EDGE_BOTH, 20000);
    if (!lcd || !g_out || !btn) {               /* 이것들이 없으면 시작하지 않는다 → systemd 가 재시작 */
        fprintf(stderr, "LCD · 출력 · 버튼 요청 실패 — overlay(eio) 를 내렸는지, 다른 프로그램이 쓰는지 확인\n");
        return EXIT_FAILURE;
    }
    sonar_t *so = sonar_open(23, 24);           /* 센서는 없어도 계속 — 화면에 -- */
    if (!so) log_msg("HC-SR04 요청 실패%s — 거리 없이 계속", "");
    lcd_printf(lcd, 0, "envmon start");
    log_msg("envmon 시작 · CSV %s", g_csv);

    pthread_t th[4];
    pthread_create(&th[0], NULL, sensor_thread, NULL);
    pthread_create(&th[1], NULL, proximity_thread, so);
    pthread_create(&th[2], NULL, control_thread, btn);
    pthread_create(&th[3], NULL, display_thread, lcd);

    while (!g_stop) sleep_ms(200);              /* 메인은 신호만 기다린다 */
    pthread_mutex_lock(&S.mx); pthread_cond_broadcast(&S.cv); pthread_mutex_unlock(&S.mx);
    for (int i = 0; i < 4; i++) pthread_join(th[i], NULL);

    enum gpiod_line_value off[4] = { 0, 0, 0, 0 };            /* 릴레이 · 부저 반드시 끄기 */
    gpiod_line_request_set_values(g_out, off);
    gpiod_line_request_release(g_out);
    gpiod_line_request_release(btn);
    sonar_close(so);
    lcd_printf(lcd, 0, "envmon stopped");
    lcd_printf(lcd, 1, "%s", "");
    lcd_release(lcd);                           /* 화면은 남기고 라인만 반납 */
    log_msg("envmon 종료%s", "");
    return EXIT_SUCCESS;
}

else if (down_at) {
    long long ms = (ts - down_at) / 1000000;
    if (ms >= 1000) S.d.mute = !S.d.mute;              /* 길게 */

    pthread_cond_signal(&S.cv);                         /* 화면 즉시 */

7-1. systemd 서비스 — 부팅하면 저절로

src/envmon.service

# envmon.service — 부팅하면 envmon 을 자동으로 실행
#
#   1) 아래 pi 와 /home/pi 를 내 사용자 이름 · 홈으로 바꾼다 (whoami)
#   2) sudo cp envmon.service /etc/systemd/system/
#   3) sudo systemctl daemon-reload
#   4) sudo systemctl enable --now envmon
#   5) journalctl -u envmon -f
[Unit]
Description=EIO smart environment monitor (Day 5)

[Service]
Type=simple
User=pi
WorkingDirectory=/home/pi/eio/day5/src
ExecStart=/home/pi/eio/day5/src/envmon -c /home/pi/eio/day5/src/envmon.csv
# systemctl stop → SIGTERM → envmon 의 핸들러가 출력을 끄고 GPIO 를 반납한다
KillSignal=SIGTERM
TimeoutStopSec=5
# 비정상 종료(코드 ≠ 0, 신호로 죽음) 때만 3초 뒤 다시 시작
Restart=on-failure
RestartSec=3

[Install]
WantedBy=multi-user.target

whoami                                       # 내 사용자 이름 → service 의 pi · /home/pi 를 바꾼다
nano envmon.service
sudo cp envmon.service /etc/systemd/system/
sudo systemctl daemon-reload                 # 새 unit 파일 읽기
sudo systemctl enable --now envmon           # 부팅 때 자동 + 지금 시작
systemctl status envmon                      # active (running)
journalctl -u envmon -f                      # 로그 따라 보기 (Ctrl+C 로 빠져나옴 — 서비스는 계속)
설정 뜻
User=pi root 가 아닌 내 계정 — gpio · i2c 그룹 권한으로 충분 (Day 1)
WorkingDirectory= CSV 가 생기는 곳
Restart=on-failure · RestartSec=3 비정상 종료 · 신호로 죽으면 3초 뒤 다시 시작
KillSignal=SIGTERM systemctl stop 이 보내는 신호 → envmon 의 핸들러가 출력 끄고 반납
WantedBy=multi-user.target enable 하면 부팅 순서에 들어감

ExecStop 을 따로 쓰지 않는 이유: 기본 종료 방법이 SIGTERM 이고, envmon 이 SIGTERM 을 받아 스스로 정리한다(Day 2 gpio_install_sigint). 정리를 프로그램 밖 스크립트에 맡기면, 프로그램이 이미 죽은 뒤라 무엇을 켜 두었는지 모른다.

로그가 안 보이면: systemd 아래 stdout 은 파이프라서 printf 가 버퍼에 쌓인다. envmon 은 setvbuf(stdout, NULL, _IOLBF, 0) 으로 줄마다 내보낸다.
