FreeRTOS 태스크 — loop 를 여러 개로

09:00–09:50

학습 목표 - 태스크를 만들고(xTaskCreatePinnedToCore) 각각 다른 주기로 동작시킨다 - delay 와 vTaskDelay 의 의미 차이, 우선순위 · 코어 · 스택을 설명한다 - 매개변수(void *)로 같은 태스크 함수를 여러 번 재사용한다

🎬 D4-1 스케줄러 타임라인 — 두 태스크가 두 코어에서 번갈아 실행되는 모습, "쉬지 않는 태스크"가 다른 태스크를 굶기는 모습

Day 1 2교시에서 본 것처럼 setup() · loop() 는 이미 loopTask 라는 FreeRTOS 태스크 안에서 돌고 있었습니다. 오늘은 태스크를 직접 여러 개 만듭니다. 각 태스크는 자기만의 스택과 무한 루프를 가진 작은 loop() 입니다.

▶ 예제 1-1 · 두 태스크, 두 LED — 서로 다른 주기

two_tasks.ino

const int LED_R = 25, LED_G = 26;

void taskRed(void *param) {
    for (;;) {                                    // 태스크는 끝나지 않는다
        digitalWrite(LED_R, !digitalRead(LED_R));
        vTaskDelay(pdMS_TO_TICKS(300));           // 300ms 쉬는 동안 다른 태스크가 돈다
    }
}

void taskGreen(void *param) {
    for (;;) {
        digitalWrite(LED_G, !digitalRead(LED_G));
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}

void setup() {
    Serial.begin(115200);
    pinMode(LED_R, OUTPUT);
    pinMode(LED_G, OUTPUT);
    //                      함수      이름     스택   매개변수 우선순위 핸들  코어
    xTaskCreatePinnedToCore(taskRed,   "red",   2048, NULL,   1,      NULL, 1);
    xTaskCreatePinnedToCore(taskGreen, "green", 2048, NULL,   1,      NULL, 1);
}

void loop() {
    Serial.printf("loop 도 돈다 %lu\n", millis() / 1000);
    vTaskDelay(pdMS_TO_TICKS(2000));
}
// → 빨강 0.3초, 초록 1초 주기로 따로 깜빡이고, 시리얼에는 2초마다 한 줄

Day 1 5교시의 millis() 비교로도 가능했던 일이지만, 태스크로 나누면 각 동작이 자기 루프만 신경 씁니다. vTaskDelay 는 "이 태스크를 재우고 다른 태스크에 CPU 를 넘긴다"는 뜻입니다. Arduino-ESP32 에서 delay() 도 속은 vTaskDelay 라서 같은 효과지만, 태스크 안에서는 의도가 드러나는 vTaskDelay 를 씁니다.

인자 오늘 값 의미
스택 2048 바이트 (ESP32 의 FreeRTOS 는 바이트 단위). printf · String 을 쓰면 4096 이상
우선순위 1 클수록 먼저. loopTask 는 1, idle 태스크는 0
코어 1 0 은 Wi-Fi · Bluetooth 가 주로 쓰는 코어, 1 은 loopTask 코어

▶ 예제 1-2 · 우선순위와 스택 사용량

task_info.ino

TaskHandle_t hWork = NULL;

void taskWork(void *param) {
    for (;;) {
        int raw = analogRead(34);
        char buf[64];                                 // 지역 배열은 스택을 쓴다
        snprintf(buf, sizeof(buf), "pot=%d", raw);
        vTaskDelay(pdMS_TO_TICKS(100));
    }
}

void setup() {
    Serial.begin(115200);
    xTaskCreatePinnedToCore(taskWork, "work", 3072, NULL, 2, &hWork, 1);
}

void loop() {
    UBaseType_t freeStack = uxTaskGetStackHighWaterMark(hWork);   // 가장 적게 남았던 양
    Serial.printf("work 스택 여유 %u 바이트 · 코어 %d · 힙 여유 %u\n",
                  freeStack, xPortGetCoreID(), ESP.getFreeHeap());
    vTaskDelay(pdMS_TO_TICKS(1000));
}
// → work 스택 여유 2140 바이트 · 코어 1 · 힙 여유 290000

High Water Mark 는 "지금까지 스택이 가장 많이 찼을 때 남아 있던 양"입니다. 수백 바이트 이하로 떨어지면 스택을 늘립니다. 태스크를 만들 때 스택은 힙에서 가져오므로 태스크가 많을수록 getFreeHeap() 이 줄어듭니다.

▶ 예제 1-3 · ⚠ 함정 — 쉬지 않는 태스크

한 일 결과
태스크 안 for (;;) { … } 에 vTaskDelay 없음 같은 코어의 낮은 우선순위 태스크가 실행되지 못함(굶음). 설정에 따라 워치독 경고 · 재부팅
태스크 함수가 return 으로 끝남 재부팅 — 태스크는 끝내지 않거나 vTaskDelete(NULL) 로 스스로 지운다
스택 1024 에 Serial.printf Stack canary watchpoint triggered 로 재부팅 (7교시)
지역 변수 주소를 태스크 매개변수로 넘김 setup 이 끝나면 사라진 메모리를 가리킴 → 전역 · static 변수 주소를 넘긴다

큐와 뮤텍스 — 태스크끼리 주고받기

10:00–10:50

학습 목표 - 큐로 한 태스크가 만든 데이터를 다른 태스크에 안전하게 넘긴다(생산자 · 소비자) - 여러 태스크가 쓰는 자원(Serial, 공유 구조체)을 뮤텍스로 보호한다 - 전역 변수 공유와 큐 · 뮤텍스의 차이를 설명한다

🎬 D4-2 큐 · 뮤텍스 — 생산 · 소비 속도를 바꾸며 큐가 차고 비는 모습, 뮤텍스 없이 출력이 섞이는 모습

▶ 예제 2-1 · 큐 — 센서 태스크가 보내고, 표시 태스크가 받는다

queue_demo.ino

typedef struct {
    int raw;
    uint32_t t_ms;
} sample_t;

QueueHandle_t q;

void taskSensor(void *p) {                      // 생산자
    for (;;) {
        sample_t s = { analogRead(34), millis() };
        if (xQueueSend(q, &s, 0) != pdTRUE)     // 0 = 가득 차면 기다리지 않음
            Serial.println("큐 가득 — 버림");
        vTaskDelay(pdMS_TO_TICKS(100));
    }
}

void taskShow(void *p) {                        // 소비자
    sample_t s;
    for (;;) {
        xQueueReceive(q, &s, portMAX_DELAY);    // 올 때까지 잠 — CPU 를 안 쓴다
        ledcWrite(25, s.raw / 16);              // 0~4095 → 0~255
        Serial.printf("raw:%d,age:%lu\n", s.raw, millis() - s.t_ms);
    }
}

void setup() {
    Serial.begin(115200);
    ledcAttach(25, 5000, 8);
    q = xQueueCreate(8, sizeof(sample_t));      // sample_t 8개 칸
    xTaskCreatePinnedToCore(taskSensor, "sensor", 3072, NULL, 2, NULL, 1);
    xTaskCreatePinnedToCore(taskShow,   "show",   3072, NULL, 1, NULL, 1);
}

void loop() { vTaskDelay(portMAX_DELAY); }

큐는 값을 복사해서 넣고 꺼냅니다(&s 의 내용 sizeof(sample_t) 바이트). 그래서 보내는 쪽의 지역 변수가 사라져도 안전합니다. xQueueReceive(…, portMAX_DELAY) 는 데이터가 없으면 잠들어 있다가 데이터가 오면 깨어납니다 — Day 1 의 폴링 대신 기다림입니다.

▶ 예제 2-2 · 뮤텍스 — Serial 을 한 번에 한 태스크만

mutex_demo.ino

#include <DHT.h>
DHT dht(27, DHT11);
SemaphoreHandle_t serialLock;

void safe_print(const char *tag, const char *msg) {
    xSemaphoreTake(serialLock, portMAX_DELAY);   // 잠금 — 다른 태스크는 여기서 기다림
    Serial.print("[");
    Serial.print(tag);
    Serial.print("] ");
    Serial.println(msg);
    xSemaphoreGive(serialLock);                  // 해제 — 반드시 짝으로
}

void taskPot(void *p) {
    char buf[32];
    for (;;) {
        snprintf(buf, sizeof(buf), "pot=%d", analogRead(34));
        safe_print("POT", buf);
        vTaskDelay(pdMS_TO_TICKS(200));
    }
}

void taskDht(void *p) {
    char buf[32];
    for (;;) {
        float t = dht.readTemperature();
        snprintf(buf, sizeof(buf), "temp=%.1f", t);
        safe_print("DHT", buf);
        vTaskDelay(pdMS_TO_TICKS(2000));
    }
}

void setup() {
    Serial.begin(115200);
    dht.begin();
    serialLock = xSemaphoreCreateMutex();
    xTaskCreatePinnedToCore(taskPot, "pot", 3072, NULL, 1, NULL, 1);
    xTaskCreatePinnedToCore(taskDht, "dht", 4096, NULL, 1, NULL, 0);   // 다른 코어
}

void loop() { vTaskDelay(portMAX_DELAY); }
// 뮤텍스 없이: "[POT] pot=[DHT] te2041mp=24.0" 처럼 섞일 수 있다

두 태스크가 다른 코어에서 동시에 실행되면 출력 한 줄이 여러 번의 print 로 이루어져 있을 때 중간에 섞입니다. 뮤텍스는 화장실 열쇠와 같습니다 — 열쇠를 가진 태스크만 들어가고, 나오면서 반드시 돌려줍니다. Day 1 6교시의 portENTER_CRITICAL 은 아주 짧은 구간용(인터럽트까지 막음), 뮤텍스는 길어도 되는 구간용입니다.

🎬 D4-3 ISR 은 짧게 — 버튼을 누르면 ISR 이 잠깐 실행되고 태스크가 깨어나는 흐름, ISR 에서 긴 일을 할 때 다른 일이 막히는 모습

Day 1 6교시에서는 ISR 이 volatile 카운터만 올리고 loop 가 계속 들여다봤습니다(폴링). 이제는 ISR 이 태스크를 깨웁니다 — 태스크는 일이 없으면 CPU 를 전혀 쓰지 않습니다.

▶ 예제 3-1 · ISR 에서 태스크 깨우기

isr_notify.ino

const int BTN = 4, LED = 25;
TaskHandle_t hWorker = NULL;
volatile uint32_t last_isr_ms = 0;

void IRAM_ATTR onButton() {
    uint32_t now = millis();
    if (now - last_isr_ms < 200) return;       // 간단한 디바운스
    last_isr_ms = now;
    BaseType_t woken = pdFALSE;
    vTaskNotifyGiveFromISR(hWorker, &woken);   // "일어나" — 이것만 한다
    portYIELD_FROM_ISR(woken);                 // 깨운 태스크가 더 급하면 바로 전환
}

void taskWorker(void *p) {
    int count = 0;
    for (;;) {
        ulTaskNotifyTake(pdTRUE, portMAX_DELAY);   // 알림이 올 때까지 잠
        count++;
        Serial.printf("버튼 %d번 — 태스크에서 처리\n", count);   // 긴 일은 여기서
        for (int i = 0; i < 3; i++) {
            digitalWrite(LED, HIGH); vTaskDelay(pdMS_TO_TICKS(80));
            digitalWrite(LED, LOW);  vTaskDelay(pdMS_TO_TICKS(80));
        }
    }
}

void setup() {
    Serial.begin(115200);
    pinMode(LED, OUTPUT);
    pinMode(BTN, INPUT_PULLUP);
    xTaskCreatePinnedToCore(taskWorker, "worker", 3072, NULL, 3, &hWorker, 1);
    attachInterrupt(digitalPinToInterrupt(BTN), onButton, FALLING);   // 태스크 먼저, ISR 나중
}

void loop() { vTaskDelay(portMAX_DELAY); }

ISR 안에서는 FromISR 이 붙은 함수만 씁니다. Serial.printf, vTaskDelay, delay 는 태스크의 몫입니다. 이 구조를 지연 처리(deferred processing) 라고 부릅니다.

TimerHandle_t tBlink;

void onBlink(TimerHandle_t t) {                 // 타이머 서비스 태스크에서 불린다

}

void setup() {

    pinMode(26, OUTPUT);
    tBlink = xTimerCreate("blink", pdMS_TO_TICKS(500), pdTRUE, NULL, onBlink);
    //                     이름     주기                  반복   ID    콜백
    xTimerStart(tBlink, 0);
}

void loop() {
    if (Serial.available()) {
        char c = Serial.read();
        if (c == 'f') xTimerChangePeriod(tBlink, pdMS_TO_TICKS(100), 0);   // 빠르게
        if (c == 's') xTimerChangePeriod(tBlink, pdMS_TO_TICKS(1000), 0);  // 느리게
        if (c == 'x') xTimerStop(tBlink, 0);
    }
    vTaskDelay(pdMS_TO_TICKS(20));
}

변수(RAM)는 전원이 꺼지면 사라집니다. 프로그램이 들어 있는 Flash 4MB 중 약 1.5MB 를 파일 영역으로 쓰면 전원이 꺼져도 남는 파일을 만들 수 있습니다. LittleFS 는 Flash 에 맞게 설계된 작은 파일 시스템입니다.

▶ 예제 4-1 · 파일 쓰기 · 읽기 · 목록

fs_basic.ino

#include <LittleFS.h>

void list_files() {
    File root = LittleFS.open("/");
    File f = root.openNextFile();
    while (f) {
        Serial.printf("  %-20s %6u B\n", f.name(), f.size());
        f = root.openNextFile();
    }
    Serial.printf("사용 %u / 전체 %u B\n", LittleFS.usedBytes(), LittleFS.totalBytes());
}

void setup() {
    Serial.begin(115200);
    delay(500);
    if (!LittleFS.begin(true)) {               // true = 처음이면 포맷
        Serial.println("LittleFS 마운트 실패");
        return;
    }
    File f = LittleFS.open("/hello.txt", FILE_WRITE);   // "w" — 새로 쓰기(덮어쓰기)
    f.println("Hello LittleFS");
    f.printf("boot at %lu ms\n", millis());
    f.close();                                  // 닫아야 Flash 에 확실히 기록

    f = LittleFS.open("/hello.txt", FILE_READ);
    while (f.available()) Serial.write(f.read());
    f.close();

    list_files();
}

void loop() { }

#include <LittleFS.h>
#include <DHT.h>

DHT dht(27, DHT11);
const char *LOG = "/log.csv";

bool log_append(float t, int pot) {
    bool is_new = !LittleFS.exists(LOG);
    File f = LittleFS.open(LOG, FILE_APPEND);
    if (!f) return false;
    if (is_new) f.println("ms,temp,pot");              // 처음 한 번 머리줄
    f.printf("%lu,%.1f,%d\n", millis(), t, pot);
    f.close();
    return true;
}

void dump_log() {
    File f = LittleFS.open(LOG, FILE_READ);
    if (!f) { Serial.println("(로그 없음)"); return; }
    while (f.available()) Serial.write(f.read());
    f.close();
}

void setup() {
    Serial.begin(115200);
    delay(500);
    LittleFS.begin(true);
    dht.begin();
    Serial.println("d = 로그 보기, x = 로그 지우기");
}

void loop() {
    static uint32_t last = 0;
    if (millis() - last >= 5000) {                     // 5초마다 한 줄
        last = millis();
        float t = dht.readTemperature();
        if (!isnan(t)) log_append(t, analogRead(34));
    }
    if (Serial.available()) {
        char c = Serial.read();
        if (c == 'd') dump_log();
        if (c == 'x') { LittleFS.remove(LOG); Serial.println("지움"); }
    }
}

🎬 D4-5 Wi-Fi 상태 — 연결 상태가 바뀌는 과정, 안테나 유무에 따른 신호 세기와 연결 성공

ESP-32U 는 외부 안테나가 필요합니다. 기판 안테나가 없는 U 모델이라 안테나 없이는 신호가 매우 약하거나 연결이 안 됩니다. USB 를 뽑은 상태에서 U.FL 커넥터에 안테나 케이블을 수직으로 딸깍 눌러 끼웁니다. U.FL 은 여러 번 끼웠다 빼면 헐거워지므로 한 번 끼운 뒤에는 그대로 둡니다.

ESP32 는 2.4GHz 만 지원합니다(5GHz 공유기 이름은 스캔에 보이지 않음). 회사 · 학교의 아이디+비밀번호 로그인 방식(WPA2-Enterprise) 은 오늘 코드로 연결되지 않으니 스마트폰 핫스팟을 2.4GHz 로 켜서 씁니다.

▶ 예제 5-1 · 주변 Wi-Fi 스캔과 연결

wifi_connect.ino

#include <WiFi.h>

const char *SSID = "YOUR_SSID";
const char *PASS = "YOUR_PASS";

void scan() {
    int n = WiFi.scanNetworks();
    Serial.printf("주변 Wi-Fi %d개\n", n);
    for (int i = 0; i < n; i++)
        Serial.printf("  %-24s %4d dBm  ch%d\n", WiFi.SSID(i).c_str(), WiFi.RSSI(i), WiFi.channel(i));
}

bool wifi_connect(uint32_t timeout_ms) {
    WiFi.mode(WIFI_STA);                          // 스테이션(공유기에 붙는 쪽)
    WiFi.begin(SSID, PASS);
    uint32_t t0 = millis();
    while (WiFi.status() != WL_CONNECTED) {
        if (millis() - t0 > timeout_ms) return false;
        delay(250);
        Serial.print(".");
    }
    Serial.println();
    return true;
}

void setup() {
    Serial.begin(115200);
    WiFi.mode(WIFI_STA);
    scan();
    if (wifi_connect(15000)) {
        Serial.printf("연결됨  IP %s  RSSI %d dBm\n", WiFi.localIP().toString().c_str(), WiFi.RSSI());
        WiFi.setAutoReconnect(true);              // 끊기면 알아서 다시
    } else {
        Serial.println("연결 실패 — SSID · 비밀번호 · 2.4GHz · 안테나 확인");
    }
}

void loop() {
    static wl_status_t last = WL_IDLE_STATUS;
    wl_status_t s = WiFi.status();
    if (s != last) {                              // 바뀔 때만 출력
        Serial.printf("상태 %d → %d\n", last, s);
        last = s;
    }
    delay(500);
}
// → 연결됨  IP 192.168.0.23  RSSI -52 dBm
RSSI 품질
-30 ~ -60 dBm 좋음
-60 ~ -75 dBm 보통 — 웹 페이지는 됨
-80 dBm 이하 나쁨 — 끊김, 안테나 확인

while 에 타임아웃이 없으면 비밀번호가 틀렸을 때 영원히 점만 찍습니다. WiFi.status() 는 Day 1 8교시처럼 상태 값이며, 바뀔 때만 반응합니다.

#include <WiFi.h>
#include <WebServer.h>
#include <DHT.h>

const char *SSID = "YOUR_SSID";
const char *PASS = "YOUR_PASS";
WebServer server(80);

void handleRoot() {
    char html[512];
    snprintf(html, sizeof(html),
        "<!doctype html><meta charset='utf-8'>"
        "<meta name='viewport' content='width=device-width'>"
        "<h1>ESP32 센서</h1>"
        "<p>온도 %.1f °C</p><p>가변저항 %d</p>"
        "<p><a href='/'>새로고침</a></p>",

}

void handleApi() {                               // 기계가 읽기 좋은 JSON
    char json[96];
    snprintf(json, sizeof(json), "{\"temp\":%.1f,\"pot\":%d,\"ms\":%lu}",
             dht.readTemperature(), analogRead(34), millis());
    server.send(200, "application/json", json);
}

void handleNotFound() {
    server.send(404, "text/plain", "Not found");
}

void setup() {
    Serial.begin(115200);
    dht.begin();
    WiFi.begin(SSID, PASS);
    while (WiFi.status() != WL_CONNECTED && millis() < 15000) delay(250);
    Serial.printf("브라우저에서 http://%s/\n", WiFi.localIP().toString().c_str());

    server.on("/", handleRoot);                  // 경로 → 함수
    server.on("/api", handleApi);

}

void handleLog() {
    xSemaphoreTake(lock, portMAX_DELAY);
    File f = LittleFS.open("/log.csv", FILE_READ);
    if (!f) server.send(404, "text/plain", "no log");
    else { server.streamFile(f, "text/csv"); f.close(); }
    xSemaphoreGive(lock);
}

void handleClear() {
    xSemaphoreTake(lock, portMAX_DELAY);
    LittleFS.remove("/log.csv");
    xSemaphoreGive(lock);
    server.sendHeader("Location", "/");
    server.send(303);
}

void setup() {
    Serial.begin(115200);
    pinMode(LED_R, OUTPUT); pinMode(LED_G, OUTPUT); pinMode(BTN, INPUT_PULLUP);
    dht.begin();
    if (!LittleFS.begin(true)) Serial.println("LittleFS 실패");

    q = xQueueCreate(8, sizeof(reading_t));       // 큐 · 뮤텍스 먼저
    lock = xSemaphoreCreateMutex();

    xTaskCreatePinnedToCore(taskSensor, "sensor", 4096, NULL, 2, NULL, 1);
    xTaskCreatePinnedToCore(taskLogger, "logger", 4096, NULL, 1, NULL, 1);
    xTaskCreatePinnedToCore(taskButton, "button", 3072, NULL, 3, &hButton, 1);
    attachInterrupt(digitalPinToInterrupt(BTN), onButton, FALLING);

    WiFi.mode(WIFI_STA);
    WiFi.begin(SSID, PASS);
    uint32_t t0 = millis();
    while (WiFi.status() != WL_CONNECTED && millis() - t0 < 15000) delay(250);
    if (WiFi.status() == WL_CONNECTED)
        Serial.printf("http://%s/\n", WiFi.localIP().toString().c_str());
    else {
        WiFi.mode(WIFI_AP);                       // 실패하면 AP 모드로 (5교시)
        WiFi.softAP("ESP32-MONITOR", "12345678");
        Serial.println("AP 모드: ESP32-MONITOR 에 접속 → http://192.168.4.1/");
    }

    server.on("/", handleRoot);
    server.on("/api", handleApi);
    server.on("/log.csv", handleLog);
    server.on("/clear", handleClear);
    server.begin();
}

void loop() {
    server.handleClient();
    vTaskDelay(pdMS_TO_TICKS(2));
